43 bool ArgList::add_arg(ArgType type, void const* data, size_t length) {
44
45 // Not enough space to add more args
46 if (m_entry_count >= MAX_ARGS || m_payload_size + length > sizeof(m_payload))
47 return false;
48
49 // Create the entry
50 m_entries[m_entry_count].type = type;
51 m_entries[m_entry_count].offset = m_payload_size;
52 m_entries[m_entry_count].length = length;
53
54 // Copy the data into the payload
55 memcpy(&m_payload[m_payload_size], data, length);
56 m_payload_size += length;
57 m_entry_count++;
58
59 return true;
60 }

344 size_t ArgList::serialise(uint8_t* buffer, size_t size) {
345
346 size_t offset = 0;
347 size_t space_needed = sizeof(m_entry_count) + m_entry_count * sizeof(arg_entry_t) + payload_size();
348
349 // Check bounds
350 if (space_needed > size)
351 return 0;
352
353 // Arg count
354 memcpy((void*) (buffer + offset), &m_entry_count, sizeof(m_entry_count));
355 offset += sizeof(m_entry_count);
356
357 // Arg entries
358 memcpy((void*) (buffer + offset), &m_entries, m_entry_count * sizeof(arg_entry_t));
359 offset += m_entry_count * sizeof(arg_entry_t);
360
361 // Payload data
362 memcpy((void*) (buffer + offset), &m_payload, m_payload_size);
363 offset += m_payload_size;
364
365 return offset;
366 }

449 bool rpc_call(const char* server, const char* function, ArgList* args, ArgList* return_values, size_t flags) {
450
451 // Open the server
452 uint64_t endpoint = open_endpoint(server);
453 while (endpoint == 0 && flags & (size_t) RPCMEssageFlags::WAIT_FOR_SERVER)
454 endpoint = open_endpoint(server);
455
456 if (endpoint == 0)
457 return false;
458
459
460 // Set up a response endpoint if not already done
461 string response_end_name = "response_"s + string(pid());
462 if (m_response_endpoint == 0)
463 m_response_endpoint = create_endpoint(response_end_name.c_str());
464
465 // Still no response endpoint
466 if (m_response_endpoint == 0) {
467 klog("ERR: cant create response endpoint\n");
468 return false;
469 }
470
471 // Add the header to the message
472 rpc_header_t header;
473 header.source_pid = pid();
474 header.flags = flags;
475 header.type = RPCMessageType::CALL;
476
477 // Create a message: name, response endpoint, args
478 ArgList message;
479 message.push_blob(&header, sizeof(RPCHeader));
480 message.push_string(function);
481 message.push_string(response_end_name);
482 message.append_args(*args);
483
484 // Send the message
485 uint8_t buffer[MAX_SERIALIZED_SIZE];
486 size_t size = message.serialise(buffer, sizeof(buffer));
487 send_message(endpoint, buffer, size);
488
489 // Read the response
490 if (!(flags & (size_t) RPCMEssageFlags::ONE_WAY)) {
491 size = read_message(m_response_endpoint, buffer, sizeof(buffer));
492 return_values->deserialise(buffer, size);
493 }
494
495 return true;
496 }

507 bool rpc_server_process_next(uint64_t endpoint, bool block) {
508
509 // Read the message
510 ArgList message;
511 uint8_t buffer[MAX_SERIALIZED_SIZE];
512 size_t size = read_message(endpoint, buffer, sizeof(buffer), block);
513
514 // No messages
515 if (!block && size == 0)
516 return false;
517
518 // Load the message header
519 message.deserialise(buffer, size);
520 auto header = (rpc_header_t*) message.get_blob(0);
521
522 // Get the function and response endpoint
523 function_entry_t function_entry = find_function(message.get_string(1).c_str());
524 uint64_t response_endpoint = open_endpoint(message.get_string(2).c_str());
525
526 // Validate request
527 if (response_endpoint == 0 || function_entry.function == nullptr)
528 return true;
529
530 // Extract the arguments
531 ArgList args;
532 args.append_args(message, 3);
533
534 // Delegate to the function
535 ArgList return_values;
536 function_entry.function(&args, &return_values);
537
538 // Send the response
539 if (!(header->flags & (size_t) RPCMEssageFlags::ONE_WAY)) {
540 size = return_values.serialise(buffer, sizeof(buffer));
541 send_message(response_endpoint, (void*) buffer, size);
542 }
543
544 return true;
545 }

592 [[noreturn]] void rpc_server_loop(Vector<uint64_t> endpoints) {
593
594 // Server loop
595 while (true) {
596
597 bool did_work = false;
598
599 // Try handle messages on each valid endpoint
600 for (const auto& endpoint : endpoints)
601 if (endpoint != 0)
602 did_work = did_work || rpc_server_process_next(endpoint, false);
603
604 // Dont hog the CPU
605 if (!did_work)
606 yield();
607 }
