114 const string this_path = m_parent.empty() ? m_name : m_parent + "/" + m_name;
115 Map<string, bool> seen_dirs;

118 const string& path = "/"s + header->filename;

120 // Only care about storing files in this directory

123

125 string rest = path.substring(this_path.length(), path.length() - this_path.length());
126
127 // File
128 if (!rest.contains('/')) {
129 m_files.push_back(new TARFile(m_volume, header));
130 continue;
131 }
132

135 auto found = seen_dirs.find(subdir);

137 continue;
