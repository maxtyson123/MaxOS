235void Directory::debug_contents(uint64_t max_depth) {
236
237 struct StackEntry {
238 Directory* dir;
239 uint64_t depth;
240 };
241
242 // Small stack container (uses your kernel Vector)
243 Vector<StackEntry> stack;
244 stack.push_back({ this, 0 });
245
246 while (!stack.empty()) {
247 StackEntry entry = stack.pop_back();
248
249 Directory* cur = entry.dir;
250 uint64_t cur_depth = entry.depth;
251 if (!cur) continue;
252
253 // Build indent string (two spaces per depth)
254 string indent;
255 for (uint64_t i = 0; i < cur_depth; ++i)
256 indent += " ";
257
258 // Print directory header
259 KPI::klog("%s[DIR ] %s\n", indent.c_str(), cur->name().c_str());
260
261 // Print files in the directory (non-recursive)
262 for (auto *f : cur->m_files) {
263 if (!f) continue;
264 KPI::klog("%s [FILE] %s\n", indent.c_str(), f->name().c_str());
265 }
266
267 // Depth limit check: if we've reached max_depth, do not push children
268 if (max_depth != (uint64_t)-1 && cur_depth >= max_depth)
269 continue;
270
271 // Push subdirectories onto the stack.
272 // Push in reverse order so the output order matches the natural forward order.
273 size_t sub_count = cur->m_subdirectories.size();
274 for (size_t i = sub_count; i-- > 0; )
275 {
276 Directory* sd = cur->m_subdirectories[i];
277 if (!sd) continue;
278 stack.push_back({ sd, cur_depth + 1 });
279 }
280 }

357Directory* FileSystem::get_directory(const string &path) {
358
359 // Check if the path is the root
360 if (path == "/")
361 return root_directory();
362
363 // Recursively open the directory
364 Directory* directory = root_directory();
365 string directory_path = path;
366 while (directory_path.length() > 0) {
367
368 // Get the name of the directory
369 string directory_name = Path::top_directory(directory_path);
370
371 // Open the directory
372 Directory* subdirectory = directory->open_subdirectory(directory_name);
373 if (!subdirectory)
374 return nullptr;
375
376 // Set the new directory
377 directory = subdirectory;
378
379 // Get the path to the next directory
380 directory_path = directory_path.substring(directory_name.length() + 1, directory_path.length() - directory_name.length() - 1);
381 }
382
383 return directory;
384}
