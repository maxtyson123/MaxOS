332common::Vector<String> String::split(String const& delimiter) const {
333 common::Vector<String> strings;
334
335 // Go through the string and split it by the delimiter
336 size_t start = 0;
337 for (size_t i = 0; i <= m_length - delimiter.length(); i++) {
338
339 // Check if matches at this position
340 bool matches = true;
341 for (size_t j = 0; j < delimiter.length(); j++)
342 if (m_string[i + j] != delimiter[j]) {
343 matches = false;
344 break;
345 }
346
347 if (!matches)
348 continue;
349
350 // Add the splice of the string
351 strings.push_back(substring(start, i - start));
352 start = i + delimiter.length();
353 i += delimiter.length() - 1;
354 }
355
356 // Add the last string to the vector
357 strings.push_back(substring(start, m_length - start));
