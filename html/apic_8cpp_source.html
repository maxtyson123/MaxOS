21LocalAPIC::LocalAPIC() {
22
23 // Get the APIC base address
24 uint64_t msr_info = CPU::read_msr(0x1B);
25 m_apic_base = msr_info & 0xFFFFF000;
26
27 // Reserve the base address once
28 static bool bsp_setup = false;
29 if(!bsp_setup)
30 PhysicalMemoryManager::s_current_manager->reserve(m_apic_base);
31
32 // Check if the APIC supports x2APIC
33 uint32_t ignored, xleaf, x2leaf;
34 CPU::cpuid(0x01, &ignored, &ignored, &x2leaf, &xleaf);
35
36 if (x2leaf & (1 << 21)) {
37
38 // Enable x2APIC
39 m_x2apic = true;
40 msr_info |= (1 << 10);
41 CPU::write_msr(0x1B, msr_info);
42 Logger::DEBUG() << "CPU supports x2APIC\n";
43
44 } else if (xleaf & (1 << 9)) {
45
46 m_x2apic = false;
47 Logger::DEBUG() << "CPU supports xAPIC\n";
48
49 // Map the APIC base address to the higher half
50 m_apic_base_high = (uint64_t) PhysicalMemoryManager::to_io_region(m_apic_base);
51 PhysicalMemoryManager::s_current_manager->map((physical_address_t*) m_apic_base, (virtual_address_t*) m_apic_base_high, PRESENT | WRITE);
52 Logger::DEBUG() << "APIC Base: phy=0x" << m_apic_base << ", virt=0x" << m_apic_base_high << "\n";
53 } else {
54 ASSERT(false, "CPU does not support xAPIC");
55 }
56
57 if (!(msr_info & (1 << 11))) {
58 Logger::WARNING() << "APIC is not enabled\n";
59 return;
60 }
61
62 // Enable the APIC
63 write(0xF0, (1 << 8) | 0x100);
64 bsp_setup = true;
65 Logger::DEBUG() << "APIC Enabled\n";
66}

132void LocalAPIC::send_icr(uint32_t icr_low, uint32_t icr_high, bool use_shorthand) const {
133
134 if (!m_x2apic) {
135
136 // Select target core
137 if (!use_shorthand)
138 write(0x310, icr_high << 24);
139
140 // Send the interrupt
141 write(0x300, icr_low);
142
143 // Wait for delivery
144 while (read(0x300) & (1 << 12))
145 asm volatile("pause");

147 } else {
148

151 if (!use_shorthand)
152 icr |= (uint64_t)icr_high << 32;
153
154 CPU::write_msr(0x830, icr);
155 }

213void LocalAPIC::send_interrupt(uint8_t apic_id, uint8_t vector, size_t flags) const {
214
215 uint32_t icr_low = 0;
216 uint32_t icr_high = 0;
217
218 // Vector (bits 0–7)
219 icr_low |= vector;
220
221 // Destination
222 bool use_shorthand = true;
223 if (flags & SEND_ALL) {
224 // All including self (11b)
225 icr_low |= (0b11 << 18);
226 } else if (flags & EXCLUDE_SELF) {
227 // All excluding self (10b)
228 icr_low |= (0b10 << 18);
229 } else {
230 // Specific core
231 icr_high = ((uint32_t)apic_id) << 24;
232 use_shorthand = false;
233 }
234
235 // Send the vector
236 send_icr(icr_low, icr_high, use_shorthand);

269 Logger::DEBUG() << "IO APIC Max Redirection Entry: 0x" << (uint64_t) m_max_redirect_entry << "\n";
270
271 // Get the source override item
272 MADTEntry* source_override_item = get_madt_item(MADT_TYPE::IO_APIC, m_override_array_size);
273 uint32_t total_length = sizeof(MADT);
274 while (total_length < m_madt->header.length && m_override_array_size < 0x10) { // 0x10 is the max items
275
276 total_length += source_override_item->length;
277
278 // If there is an override, populate the array
279 if (source_override_item != nullptr && source_override_item->type == 2) {
280
281 // Get the override and populate the array
282 auto* override = (Override*) (source_override_item + 1);
283 m_override_array[m_override_array_size].bus = override->bus;
284 m_override_array[m_override_array_size].source = override->source;
285 m_override_array[m_override_array_size].global_system_interrupt = override->global_system_interrupt;
286 m_override_array[m_override_array_size].flags = override->flags;
287 m_override_array_size++;
288 }
289
290 // Get the next item
291 source_override_item = get_madt_item(MADT_TYPE::IO_APIC, m_override_array_size);

308MADTEntry* IOAPIC::get_madt_item(MADT_TYPE type, uint8_t index) {
309
310 // The item starts at the start of the MADT
311 auto* item = (MADTEntry*) ((uint64_t) m_madt + sizeof(MADT));
312 uint64_t total_length = 0;
313 uint8_t current_index = 0;
314
315 // Loop through the items
316 while (total_length + sizeof(MADT) < m_madt->header.length && current_index <= index) {
317
318 // Correct type
319 if (item->type == (uint8_t)type) {
320
321 // Correct index means found
322 if (current_index == index)
323 return item;
324
325 // Right type wrong index so move on
326 current_index++;
327 }
328
329 // Increment the total length
330 total_length += item->length;
331 item = (MADTEntry*) ((uint64_t) item + item->length);
332 }
333
334 // No item found
335 return nullptr;
336}

412void IOAPIC::set_redirect(interrupt_redirect_t* redirect) {
413
414 // Create the redirection entry
415 RedirectionEntry entry = {};
416 entry.raw = redirect->flags | (redirect->interrupt & 0xFF);
417 entry.destination = redirect->destination;
418 entry.mask = redirect->mask;
419
420 // Check if a global system interrupt is used
421 for (uint8_t i = 0; i < m_override_array_size; i++) {
422
423 if (m_override_array[i].source != redirect->type)
424 continue;
425
426 // Set the lower 4 bits of the pin
427 entry.pin_polarity = ((m_override_array[i].flags & 0b11) == 2) ? 0b1 : 0b0;
428 entry.pin_polarity = (((m_override_array[i].flags >> 2) & 0b11) == 2) ? 0b1 : 0b0;
429
430 // Set the trigger mode
431 entry.trigger_mode = (((m_override_array[i].flags >> 2) & 0b11) == 2);
432 break;
433
434 }
435
436 // Write the redirect
437 write_redirect(redirect->index, &entry);
438}
