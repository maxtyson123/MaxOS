91void FramebufferConsole::put_character(uint16_t x, uint16_t y, char c) {
92
93 // Parse any ansi codes
94 if(c == '\033') {
95
96 // Store the character
97 ansi_code_length = 0;
98 ansi_code[ansi_code_length++] = c;
99
100 // Do not draw the escape character
101 return;
102 }
103
104 if(ansi_code_length < 8) {
105
106 // Add the character to the ANSI code
107 ansi_code[ansi_code_length++] = c;
108
109 // If the ANSI code is complete
110 if(c == 'm') {
111 ansi_code[ansi_code_length] = '\0';
112 ansi_code_length = -1;
113
114 if(strcmp("\033[0m", ansi_code) == 0) {
115 m_foreground_color = ConsoleColour::Uninitialised;
116 m_background_color = ConsoleColour::Uninitialised;
117 return;
118 }
119
120 // Get the colour from the ANSI code
121 const Colour colour(ansi_code);
122
123 // Set the colour
124 bool foreground = ansi_code[4] == '3';
125 if(foreground)
126 m_foreground_color = colour.to_console_colour();
127 else
128 m_background_color = colour.to_console_colour();
129
130 }
131
132 // Do not draw the escape character
133 return;
134 }
135
136 // If the coordinates are out of bounds, return
137 if(x >= width() || y >= height())
138 return;
139
140 // Set the character at the offset, by masking the character with the current character (last 8 bits)
141 int offset = (y * width() + x);
142 m_video_memory_meta[offset] = (m_video_memory_meta[offset] & 0xFF00) | (uint16_t) c;
143
144 // Convert the char into a string
145 char s[] = " ";
146 s[0] = c;
147
148 // Get the colour as a number
149 Colour foreground = m_foreground_color == ConsoleColour::Uninitialised ? Colour(get_foreground_color(x, y)) : Colour(m_foreground_color);
150 Colour background = m_background_color == ConsoleColour::Uninitialised ? Colour(get_background_color(x, y)) : Colour(m_background_color);
151 auto fg = GraphicsContext::colour_to_int(foreground, m_framebuffer_bpp);
152 auto bg = GraphicsContext::colour_to_int(background, m_framebuffer_bpp);
153
154 // Draw the text from top to bottom
155 for(int y_bit_map_offset = 0; y_bit_map_offset < FONT_WIDTH; y_bit_map_offset++) {
156 for(int x_bit_map_offset = 0; x_bit_map_offset < FONT_HEIGHT - (2 * FONT_PADDING); ++x_bit_map_offset) {
157
158 // Get the character
159 uint8_t character = c;
160
161 // Draw the pixel
162 bool set = AMIGA_FONT[(uint16_t) character * 8 + y_bit_map_offset] & (128 >> (x_bit_map_offset % 8));
163 put_pixel(x * 8 + x_bit_map_offset, y * FONT_HEIGHT + y_bit_map_offset, set ? fg : bg);
164
165 }
166 }
167}

326void FramebufferConsole::scroll_up(uint16_t left, uint16_t top, uint16_t width,
327 uint16_t height,
328 common::ConsoleColour foreground,
329 common::ConsoleColour background, char fill) {
330
331
332 // Get the framebuffer info
333 uint64_t framebuffer_width = s_framebuffer_info->common.framebuffer_width;
334 uint64_t framebuffer_bpp = m_framebuffer_bpp; // in bits per pixel
335 uint64_t bytes_per_pixel = framebuffer_bpp / 8;
336 uint64_t framebuffer_pitch = framebuffer_width * bytes_per_pixel;
337
338 uint16_t line_height = FONT_HEIGHT;
339
340 // Region conversions
341 uint16_t region_pixel_y = top * line_height;
342 uint16_t region_pixel_height = height * line_height;
343 uint16_t region_pixel_left = left * FONT_WIDTH;
344 uint16_t region_pixel_width = width * FONT_WIDTH;
345 size_t row_bytes = region_pixel_width * bytes_per_pixel;
346
347 // Decide the colour of the pixel
348 ConsoleColour to_set_foreground = get_foreground_color(left, top + height - 1);
349 ConsoleColour to_set_background = get_background_color(left, top + height - 1);
350 uint32_t fill_value = GraphicsContext::colour_to_int(Colour(to_set_background), m_framebuffer_bpp);
351
352 // Scroll the region upward by one text line
353 for(uint16_t row = 0; row < region_pixel_height - line_height; row++) {
354 uint8_t* src = (uint8_t*)m_framebuffer_address + (region_pixel_y + row + line_height) * framebuffer_pitch + region_pixel_left * bytes_per_pixel;
355 uint8_t* dest = (uint8_t*)m_framebuffer_address + (region_pixel_y + row) * framebuffer_pitch + region_pixel_left * bytes_per_pixel;
356 memmove(dest, src, row_bytes);
357 }
358
359 // Clear the last line of the region
360 uint16_t clear_start_y = region_pixel_y + region_pixel_height - line_height;
361 for(uint16_t row = 0; row < line_height; row++) {
362 auto row_add = (uint32_t*) ((uint8_t*)m_framebuffer_address + (clear_start_y + row) * framebuffer_pitch + region_pixel_left * 4);
363 for(uint16_t col = 0; col < region_pixel_width; col++) {
364 row_add[col] = fill_value;
365 }
366 }
367
368 //Update any per-pixel colour metadata
369 uint16_t text_row = top + height - 1;
370 for(uint16_t x = left; x < left + width; x++) {
371 set_foreground_color(x, text_row, to_set_foreground);
372 set_background_color(x, text_row, to_set_background);
373 }
374}

392void FramebufferConsole::update_progress_bar(uint8_t percentage) {
393
394 // Not ready
395 if(!s_framebuffer_info)
396 return;
397
398 // Check bounds
399 if(percentage > 100)
400 percentage = 100;
401
402 uint8_t progress_height = 15;
403 uint8_t progress_spacing = 20;
404 uint8_t progress_width_cull = 40;
405
406 // Find the center of the screen
407 uint32_t right_x = (s_framebuffer_info->common.framebuffer_width / 2) - LOGO_WIDTH / 2;
408 uint32_t bottom_y = (s_framebuffer_info->common.framebuffer_height / 2 - 80) - LOGO_HEIGHT / 2;
409
410 // Find the bounds
411 uint32_t start_x = progress_width_cull;
412 uint32_t start_y = LOGO_HEIGHT + progress_spacing;
413 uint32_t end_x = LOGO_WIDTH - progress_width_cull;
414 uint32_t end_y = LOGO_HEIGHT + progress_height + progress_spacing;
415
416 // Draw the progress bar
417 for(uint32_t progress_y = start_y; progress_y < end_y; ++progress_y) {
418 for(uint32_t progress_x = start_x; progress_x < end_x; ++progress_x) {
419
420 // Check if drawing border
421 bool is_border = (progress_y == start_y) || (progress_y == end_y - 1) ||
422 (progress_x == start_x) || (progress_x == end_x - 1);
423
424 // Only draw the border if it is the first time drawing it
425 is_border = is_border && percentage == 0;
426
427 // If it is not within the percentage, skip it
428 uint32_t logo_progress_width = LOGO_WIDTH * percentage / 100;
429 if(progress_x > logo_progress_width && !is_border)
430 continue;
431
432 // Draw the pixel
433 auto col = GraphicsContext::colour_to_int(Colour(0xFF, 0xFF, 0xFF), s_framebuffer_info->common.framebuffer_bpp);
434 put_pixel(right_x + progress_x, bottom_y + progress_y, col);
435
436 }
437 }
