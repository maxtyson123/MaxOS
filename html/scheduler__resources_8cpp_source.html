46int ProcessResource::read(void* buffer, size_t size, size_t flags) {
47
48 // Process not found (allow reading stats only to get exit code)
49 if(!process && flags != (size_t)ProcessFlags::READ_STATS)
50 return -1;
51
52 switch ((ProcessFlags)flags) {
53
54 case ProcessFlags::READ_STATS: {
55
56 // Check bounds
57 if(size < sizeof(process_stats_t))
58 return -1;
59
60 // Get the info
61 process_stats_t stats = {};
62 if(process) {
63 stats.pid = process->pid();
64 stats.parent_pid = 0; //TODO: store parent pid
65 stats.thread_count = process->threads().size();
66 stats.total_ticks = process->total_ticks();
67 stats.memory_usage = 0; //process->memory_manager->memory_used(); TODO: boken?
68 stats.handle_count = process->resource_manager.resources().size();
69 }
70 stats.exit_code = exit_code;
71
72 // Copy across
73 memcpy(buffer, &stats, sizeof(process_stats_t));
74 return sizeof(process_stats_t);
75
76 }
77
78 case ProcessFlags::READ_ENVIRONMENT: {
79
80 // Check bounds
81 if(size < sizeof(process_environment_t))
82 return -1;
83
84 // Get the info
85 process_environment_t info;
86 info.args = 0;
87 info.arg_count = 0; //TODO: store
88 info.env_vars = 0; //TODO: implement
89 info.env_var_count = 0;
90 info.working_directory = (const char*)MemoryManager::malloc(process->working_directory.length() + 1);
91 info.executable_path = 0;// TODO: store
92
93 // Copy the info
94 // .. args
95 // .. env vars
96 memcpy((void*)info.working_directory, process->working_directory.c_str(), process->working_directory.length() + 1);
97 // .. executable path
98
99 // Copy across
100 memcpy(buffer, &info, sizeof(process_environment_t));
101 return sizeof(process_environment_t);
102
103 }
104
105
106 case ProcessFlags::READ_HANDLES: {
107
108 // Check bounds
109 size_t size_needed = sizeof(handle_info_t) * process->resource_manager.resources().size();
110 if(size < size_needed)
111 return -1;
112
113 for(const auto& resource : process->resource_manager.resources()) {
114
115 // Get the info
116 handle_info_t handle_info;
117 handle_info.handle = resource.first;
118 handle_info.type = resource.second->type();
119 handle_info.name = (const char*)MemoryManager::malloc(resource.second->name().length() + 1);
120 memcpy((void*)handle_info.name, resource.second->name().c_str(), resource.second->name().length() + 1);
121
122 // Copy across
123 memcpy(buffer, &handle_info, sizeof(handle_info_t));
124 buffer = (void*)((uintptr_t)buffer + sizeof(handle_info_t));
125 }
126
127 return (int)size_needed;
128
129 }
130
131 default:
132 return -1;
133 }
134}

145int ProcessResource::write(const void* buffer, size_t size, size_t flags) {
146
147 // Process not found
148 if(!process)
149 return -1;
150
151 switch ((ProcessFlags)flags) {
152
153 case ProcessFlags::WRITE_CHANGE_DIRECTORY: {
154
155 // Change the working directory
156 string new_directory = string((const char*)buffer);
157 process->working_directory = new_directory;
158 return size;
159 }
160
161 case ProcessFlags::WRITE_KILL: {
162
163 // Kill the process
164 exit_code = *((uint64_t*)buffer);
165 GlobalScheduler::remove_process(process);
166 process->threads()[0]->yield();
167
168 return size;
169 }
170
171 default:
172 return -1;
173 }

252int ThreadResource::write(const void* buffer, size_t size, size_t flags) {
253
254 // Thread not found
255 if(!thread)
256 return -1;
257
258 switch ((ThreadFlags)flags) {
259
260 case ThreadFlags::WRITE_SLEEP_TIME: {
261
262 // Set the sleep time
263 uint64_t sleep_time = *((uint64_t*)buffer);
264 thread->sleep(sleep_time);
265 return size;
266 }
267
268 case ThreadFlags::WRITE_KILL: {
269
270 // Kill the thread
271 exit_code = *((uint64_t*)buffer);
272 thread->thread_state = ThreadState::STOPPED;
273 thread->yield();
274 return size;
275 }
276
277 case ThreadFlags::WRITE_YIELD: {
278 thread->yield();
279 return size;
280 }
281
282 case ThreadFlags::WRITE_ENABLE_PORT_IO: {
283 uint64_t port = *((uint64_t*)buffer);
284 thread->enable_port(port);
285 return size;
286 }
287
288 case ThreadFlags::WRITE_DISABLE_PORT_IO: {
289 uint64_t port = *((uint64_t*)buffer);
290 thread->disable_port(port);
291 return size;
292 }
293
294 default:
295 return -1;
296 }

362Resource* ProcessResourceRegistry::create_resource(string const& name, size_t flags, uintptr_t data) {
363
364 // Get the command
365 auto command = (execute_command_t*)data;
366 if (!command)
367 return nullptr;
368
369 // Kernel copy (@todo copy the data page into the new process so dont get unmapped error and also thus dont need a copy)
370 auto exec_buffer = buffer_t(command->file_data_size);
371 exec_buffer.copy_from(command->file_data, command->file_data_size);
372
373 // Copy the args into userspace (@todo bad, two copies)
374 uint64_t argc = command->args_count;
375 size_t arg_size = argc * sizeof(void*);
376 void* argv = MemoryManager::kmalloc(arg_size);
377 for (int i = 0; i < argc; i++) {
378
379 // Copy each argument
380 size_t len = strlen(((char**)command->args)[i]) + 1;
381 ((char**)argv)[i] = (char*) MemoryManager::kmalloc(len);
382 memcpy((void*) ((char**)argv)[i], (void*) ((char**)command->args)[i], len);
383 }
384
385 // Load the elf
386 auto elf = new ELF64((uintptr_t)exec_buffer.raw());
387 if (!elf->is_valid())
388 return nullptr;
389
390 // Create the processs
391 auto process = new Process(name, argv, argc, elf);
392 GlobalScheduler::system_scheduler() -> add_process(process);
393
394 // Create the resource
395 auto resource = new ProcessResource(name, flags, resource_type_t::PROCESS);
396 resource->process = process;
397
398 // Clean up
399 // for (int i = 0; i < argc; i++)
400 // delete ((char**)argv)[i];
401 // delete argv;
402
403
404 // Switch back to the caller proc
405 MemoryManager::switch_active_memory_manager(GlobalScheduler::current_process()->memory_manager);
406 return resource;
