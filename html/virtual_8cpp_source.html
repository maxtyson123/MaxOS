25VirtualMemoryManager::VirtualMemoryManager() {
26
27 // Set the kernel flag
28 bool is_kernel = MemoryManager::s_kernel_memory_manager == nullptr;
29 if (!is_kernel) {
30
31 // Get a new pml4 table
32 m_pml4_root_physical_address = (uint64_t*)PhysicalMemoryManager::s_current_manager->allocate_frame();
33 m_pml4_root_address = (uint64_t*)PhysicalMemoryManager::to_dm_region((uint64_t)m_pml4_root_physical_address);
34
35 // Clear the table
36 PhysicalMemoryManager::clean_page_table(m_pml4_root_address);
37
38 // Map the higher half of the kernel (p4 256 - 511)
39 for (size_t i = 256; i < 512; i++) {
40
41 // Recursive Map the pml4 table (so that we can access the new pml4 table later on)
42 if (i == 510) {
43 m_pml4_root_address[i] = (uint64_t)m_pml4_root_physical_address | PRESENT | WRITE;
44 continue;
45 }
46
47 // Set the new pml4 table to the old (kernel) pml4 table
48 m_pml4_root_address[i] = PhysicalMemoryManager::s_current_manager->pml4_root_address()[i];
49
50 }
51
52 // Load the userspace pml4
53 asm volatile("mov %0, %%cr3"::"r"((uint64_t)m_pml4_root_physical_address) : "memory");
54
55 } else {
56 m_pml4_root_address = PhysicalMemoryManager::s_current_manager->pml4_root_address();
57 m_pml4_root_physical_address = (uint64_t*)PhysicalMemoryManager::to_lower_region((uint64_t)m_pml4_root_address);
58 }
59
60 // Set up the first region
61 new_region();
62 m_first_region = m_current_region;
63
64 // Calculate the next available address (kernel needs to reserve space for the higher half)
65 m_next_available_address = is_kernel ? HIGHER_HALF_DIRECT_MAP + PhysicalMemoryManager::s_current_manager->memory_size() + VMM_RESERVED : PAGE_SIZE;
66}

71VirtualMemoryManager::~VirtualMemoryManager() {
72
73 // Free all the frames used by the VMM
74 virtual_memory_region_t* region = m_first_region;
75
76 // Loop through the regions
77 while (region != nullptr) {
78
79 // Loop through the chunks
80 for (size_t i = 0; i < CHUNKS_PER_PAGE; i++) {
81
82 // Have reached the end?
83 if (i == m_current_chunk && region == m_current_region)
84 break;
85
86 // Cant free shared frames that may be in use by another VMM
87 if (region->chunks[i].flags & SHARED)
88 continue;
89
90 // Free each page in the chunk
91 size_t pages = PhysicalMemoryManager::size_to_frames(region->chunks[i].size);
92 for (size_t j = 0; j < pages; j++) {
93
94 // Convert the virtual address to a physical address and free it
95 physical_address_t* frame = PhysicalMemoryManager::s_current_manager->get_physical_address((virtual_address_t*)(region->chunks[i].start_address + j * PAGE_SIZE), m_pml4_root_address);
96 if (frame != nullptr)
97 PhysicalMemoryManager::s_current_manager->free_frame(frame);
98
99 }
100 }
101
102 region = region->next;
103 }

142void* VirtualMemoryManager::allocate(uint64_t address, size_t requested_size, size_t& actual_size, size_t flags) {
143
144 // Make sure allocating something
145 if (requested_size == 0)
146 return nullptr;
147
148 // If specific address is given
149 if (address != 0) {
150
151 // Make sure isn't already allocated
152 if (address < m_next_available_address)
153 return nullptr;
154
155 // Make sure its aligned
156 if (!PhysicalMemoryManager::check_aligned(address))
157 return nullptr;
158
159 }
160
161 // Make sure the size is aligned
162 actual_size = PhysicalMemoryManager::align_up_to_page(requested_size, PAGE_SIZE);
163
164 // Check the free list for a chunk (if not asking for a specific address)
165 free_chunk_t* reusable_chunk = address == 0 ? find_and_remove_free_chunk(actual_size) : nullptr;
166 if (reusable_chunk != nullptr) {
167
168 // If the chunk is being reserved then the old memory needs to be unmapped (as the owner of this reserved region will perform the mapping as required)
169 uintptr_t chunk_start = reusable_chunk->start_address;
170 if (flags & RESERVE) {
171
172 // Unmap the memory
173 size_t pages = PhysicalMemoryManager::size_to_frames(actual_size);
174 for (size_t i = 0; i < pages; i++) {
175
176 // Free the frame
177 physical_address_t* frame = PhysicalMemoryManager::s_current_manager->get_physical_address((virtual_address_t*)(chunk_start + i * PAGE_SIZE), m_pml4_root_address);
178 if (frame != nullptr)
179 PhysicalMemoryManager::s_current_manager->free_frame(frame);
180
181 // Unmap the page
182 auto page = (virtual_address_t*)(chunk_start + (i * PAGE_SIZE));
183 if (page != nullptr)
184 PhysicalMemoryManager::s_current_manager->unmap(page, m_pml4_root_address);
185
186 }
187 }
188
189 // Ensure there is enough space if no reusable chunk is found
190 if (m_current_chunk >= CHUNKS_PER_PAGE)
191 new_region();
192
193 // Allocate the memory
194 virtual_memory_chunk_t* chunk = &m_current_region->chunks[m_current_chunk++];
195 chunk->size = actual_size;
196 chunk->flags = flags;
197 chunk->start_address = chunk_start;
198
199 // Return the address
200 return (void*)chunk_start;
201 }
202
203 // Is there space in the current region
204 if (m_current_chunk >= CHUNKS_PER_PAGE)
205 new_region();
206
207 // If needed to allocate at a specific address, fill with free memory up to that address to prevent fragmentation
208 if (address != 0)
209 fill_up_to_address(address, flags, false);
210
211 // fill_up_to_address() may have filled the last slot.
212 if (m_current_chunk >= CHUNKS_PER_PAGE)
213 new_region();
214
215 // Allocate the memory
216 virtual_memory_chunk_t* chunk = &m_current_region->chunks[m_current_chunk];
217 chunk->size = actual_size;
218 chunk->flags = flags;
219 chunk->start_address = m_next_available_address;
220
221 // Update the next available address
222 m_next_available_address += actual_size;
223 m_current_chunk++;
224
225 // If just reserving the space don't map it
226 if (flags & RESERVE)
227 return (void*)chunk->start_address;
228
229 // Map the memory
230 size_t pages = PhysicalMemoryManager::size_to_frames(actual_size);
231 for (size_t i = 0; i < pages; i++) {
232
233 // Allocate a new frame
234 physical_address_t* frame = PhysicalMemoryManager::s_current_manager->allocate_frame();
235 ASSERT(frame != nullptr, "Failed to allocate frame (from current region)\n");
236
237 // Map the frame
238 PhysicalMemoryManager::s_current_manager->map(frame, (virtual_address_t*)(chunk->start_address + i * PAGE_SIZE), PRESENT | WRITE, m_pml4_root_address);
239
240 }
241
242 // Return the address
243 return (void*)chunk->start_address;
244}

249void VirtualMemoryManager::new_region() {
250
251 // Space for the new region
252 physical_address_t* new_region_physical = PhysicalMemoryManager::s_current_manager->allocate_frame();
253 ASSERT(new_region_physical != nullptr, "Failed to allocate new VMM region\n");
254
255 // Align the new region

257
258 // Map the new region
259 PhysicalMemoryManager::s_current_manager->map(new_region_physical, (virtual_address_t*)new_region, PRESENT | WRITE, m_pml4_root_address);
260 new_region->next = nullptr;
261
262 // Clear the new region
263 for (size_t i = 0; i < CHUNKS_PER_PAGE; i++) {
264 new_region->chunks[i].size = 0;
265 new_region->chunks[i].flags = 0;
266 new_region->chunks[i].start_address = 0;
267 }
268
269 // Set the current region
270 if (m_current_region)
271 m_current_region->next = new_region;
272 m_current_chunk = 0;
273 m_current_region = new_region;
274}

282void VirtualMemoryManager::free(void* address, bool reusable_pages) {
283
284 // Make sure freeing something
285 if (address == nullptr)
286 return;
287
288 // Find the chunk
289 virtual_memory_region_t* region = m_first_region;
290 virtual_memory_chunk_t* chunk = nullptr;
291 while (region != nullptr) {
292
293 // Loop through the chunks
294 for (size_t i = 0; i < CHUNKS_PER_PAGE; i++) {
295
296 // Check if the address is in the chunk
297 if (region->chunks[i].start_address == (uintptr_t)address) {
298 chunk = &region->chunks[i];
299 break;
300 }
301 }
302
303 // If the chunk was found
304 if (chunk != nullptr)
305 break;
306
307 // Move to the next region
308 region = region->next;
309 }
310
311 // Make sure the chunk was found
312 if (chunk == nullptr)
313 return;
314
315 // No longer using a shared memory chunk
316 if (chunk->flags & VirtualFlags::SHARED) {
317
318 // Find the resource
319 for (const auto& resource : GlobalScheduler::current_process()->resource_manager.resources()) {
320 if (resource.second->type() != resource_type_t::SHARED_MEMORY)
321 continue;
322
323 // Skip shared memory that points elsewhere TODO: this is vaddr so get paddr and comapre?
324 auto shared = (SharedMemory*)resource.second;
325 if ((void*)shared->physical_address() != address)
326 continue;
327
328 // Close the resource
329 GlobalScheduler::current_process()->resource_manager.close_resource(resource.first, 0);
330 }
331 }
332
333 // Can only re use pages we own
334 if (reusable_pages && !(chunk->flags & (SHARED | RESERVE)))
335 add_free_chunk(chunk->start_address, chunk->size);
336
337 // Clear the chunk
338 chunk->size = 0;
339 chunk->flags = 0;
340 chunk->start_address = 0;
341}

441virtual_memory_chunk_t* VirtualMemoryManager::find_containing_chunk(const void* start_address, size_t size) {
442
443 // Search all the regions for the virtual chunk containing the range
444 virtual_memory_region_t* region = m_first_region;
445 while (region != nullptr) {
446 for (size_t i = 0; i < CHUNKS_PER_PAGE; i++) {
447
448 // Get the chunk info
449 auto chunk = region->chunks[i];
450 auto chunk_start = (void*)chunk.start_address;
451 auto chunk_end = (void*)(chunk.start_address + chunk.size);
452
453 // Does the region start in this chunk?
454 if (!(start_address >= chunk_start && start_address <= chunk_end))
455 continue;
456
457 // Can this chunk contain the whole region?
458 if (!(start_address + size <= chunk_end))
459 continue;
460
461 return &region->chunks[i];
462
463 }
464 region = region->next;
465 }
466
467 // Not found
468 return nullptr;
469}

552void* VirtualMemoryManager::load_range_from_process(VirtualMemoryManager* other_vmm, const void* start_address, size_t size) {
553
554 // Check bounds
555 if (!other_vmm || !start_address || size == 0)
556 return nullptr;
557
558 // Convert given region to be page orientated
559 size_t aligned_start = PhysicalMemoryManager::align_direct_to_page((size_t)start_address);
560 size_t offset = (size_t)start_address - aligned_start;
561 size_t total_size = offset + size;
562 size_t paged_size = PhysicalMemoryManager::align_up_to_page(total_size, PAGE_SIZE);
563
564 // Reserve space to receive the copy
565 auto dest = allocate(paged_size, RESERVE);
566 if (!dest)
567 return nullptr;
568
569 // Find the pages that contains the chunk
570 size_t range_pages = paged_size / PAGE_SIZE;
571
572 // Map each page in the range
573 for (size_t i = 0; i < range_pages; i++) {
574 uintptr_t virtual_page = aligned_start + i * PAGE_SIZE;
575
576 // Haven't reached the start of the region
577 if (virtual_page + PAGE_SIZE <= aligned_start)
578 continue;
579
580 // Dont copy pages past the region
581 if (virtual_page >= aligned_start + paged_size)
582 break;
583
584 // Map the page into this vmm
585 physical_address_t* frame = PhysicalMemoryManager::s_current_manager->get_physical_address((virtual_address_t*)virtual_page, other_vmm->pml4_root_address());
586 if (frame == nullptr)
587 continue;
588
589 PhysicalMemoryManager::s_current_manager->map(frame, (virtual_address_t*)(dest + virtual_page - aligned_start), PRESENT | WRITE, m_pml4_root_address);
590 }
591
592 // Return where the region actually starts, not the chunk
593 return dest + offset;
594}

602void VirtualMemoryManager::unload_range_from_process(const void* start_address, size_t size) {
603
604 // Check bounds
605 if (!start_address || size == 0)
606 return;
607
608 // Convert given region to be page orientated
609 size_t aligned_start = PhysicalMemoryManager::align_direct_to_page((size_t)start_address);
610 size_t offset = (size_t)start_address - aligned_start;
611 size_t total_size = offset + size;
612 size_t paged_size = PhysicalMemoryManager::align_up_to_page(total_size, PAGE_SIZE);
613
614 // Find the chunk that contains the region
615 auto containing_chunk = find_containing_chunk(start_address, size);
616 size_t chunk_pages = PhysicalMemoryManager::size_to_frames(containing_chunk->size);
617
618 // Unmap each page in the range
619 for (size_t i = 0; i < chunk_pages; i++) {
620 uintptr_t virtual_page = containing_chunk->start_address + i * PAGE_SIZE;
621
622 // Haven't reached the start of the region
623 if (virtual_page + PAGE_SIZE <= aligned_start)
624 continue;
625
626 // Dont unmap pages past the region
627 if (virtual_page >= aligned_start + paged_size)
628 break;
629
630 // Free the page
631 PhysicalMemoryManager::s_current_manager->unmap((virtual_address_t*)virtual_page, m_pml4_root_address);
632 }
633
634 // Free the whole region
635 free((void*)aligned_start, false);

647void VirtualMemoryManager::fill_up_to_address(uintptr_t address, size_t flags, bool mark_used) {
648
649 // Make sure the address is aligned
650 address = PhysicalMemoryManager::align_to_page(address);
651
652 // Make sure the address is not before the next available address
653 ASSERT(address >= m_next_available_address, "FAILED TO FILL: Address is before the next available address - 0x%x < 0x%x\n", address, m_next_available_address);
654
655 // Calculate the size
656 size_t size = address - m_next_available_address;
657
658 // Allocate the memory
659 virtual_memory_chunk_t* chunk = &m_current_region->chunks[m_current_chunk];
660 chunk->size = size;
661 chunk->flags = flags;
662 chunk->start_address = m_next_available_address;
663
664 // Update the next available address
665 m_next_available_address += size;
666 m_current_chunk++;
667
668 // Map the memory
669 size_t pages = PhysicalMemoryManager::size_to_frames(size);
670 for (size_t i = 0; i < pages; i++) {
671
672 // Allocate a new frame
673 physical_address_t* frame = PhysicalMemoryManager::s_current_manager->allocate_frame();
674 ASSERT(frame != nullptr, "Failed to allocate frame (from fill up)\n");
675
676 // Map the frame
677 PhysicalMemoryManager::s_current_manager->map(frame, (virtual_address_t*)(chunk->start_address + i * PAGE_SIZE), PRESENT | WRITE, m_pml4_root_address);
678
679 }
680
681 // Mark as free if needed
682 if (!mark_used)
683 free((void*)chunk->start_address);
684}
