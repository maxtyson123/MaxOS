108void Core::init_tss() {
109
110 // The reserved have to be 0
111 tss.reserved0 = 0;
112 tss.reserved1 = 0;
113 tss.reserved2 = 0;
114 tss.reserved3 = 0;
115
116 // Kernel stack (scheduler will set the threads stack)
117 tss.rsp0 = (uint64_t)m_stack + BOOT_STACK_SIZE;
118 tss.rsp1 = 0;
119 tss.rsp2 = 0;
120
121 // Interrupt stacks can all be 0 (@todo implment)
122 tss.ist1 = 0;
123 tss.ist2 = 0;
124 tss.ist3 = 0;
125 tss.ist4 = 0;
126 tss.ist5 = 0;
127 tss.ist6 = 0;
128 tss.ist7 = 0;
129
130 // All ports denied by default
131 memset(&tss.io_bitmap, 0xFF, sizeof(tss.io_bitmap));
132 tss.io_bitmap[IO_BITMAP_SIZE] = 0xFF;
133 tss.io_bitmap_offset = offsetof(tss_t, io_bitmap);
134
135 // Split the base into 4 parts (16 bits, 8 bits, 8 bits, 32 bits)
136 auto base = (uint64_t) &tss;
137 uint16_t base_1 = base & 0xFFFF;
138 uint8_t base_2 = (base >> 16) & 0xFF;
139 uint8_t base_3 = (base >> 24) & 0xFF;
140 uint32_t base_4 = (base >> 32) & 0xFFFFFFFF;
141
142 auto limit_low = (uint16_t)(sizeof(tss) - 1);
143
144 // Flags: 1 - Type = 0x9, Descriptor Privilege Level = 0, Present = 1, 2 - Available = 0, Granularity = 0
145 uint8_t flags_1 = 0x89;
146 uint8_t flags_2 = 0;
147
148 // Create the TSS descriptors
149 uint64_t tss_descriptor_low = (uint64_t) base_3 << 56 | (uint64_t) flags_2 << 48 | (uint64_t) flags_1 << 40 | (uint64_t) base_2 << 32 | (uint64_t) base_1 << 16 | (uint64_t) limit_low;
150 uint64_t tss_descriptor_high = base_4;
151
152 // Store in the GDT
153 gdt -> table[5] = tss_descriptor_low;
154 gdt -> table[6] = tss_descriptor_high;
155 gdt -> load();
156
157 // Load the TSS
158 Logger::DEBUG() << "Loading TSS: 0x0" << tss_descriptor_low << " 0x0" << tss_descriptor_high << " at 0x" << (uint64_t) &tss << "\n";
159 asm volatile("ltr %%ax" : : "a" (0x28));
160
161}

166void Core::init_sse() {
167
168 // Get the CR0 register
169 uint64_t cr0;
170 asm volatile("mov %%cr0, %0" : "=r" (cr0));
171
172 // Get the CR4 register
173 uint64_t cr4;
174 asm volatile("mov %%cr4, %0" : "=r" (cr4));
175
176 // Check if FPU is supported
177 ASSERT(CPU::check_cpu_feature(CPU_FEATURE_EDX::FPU), "FPU not supported - needed for SSE");
178
179 // Clear the emulation flag, task switch flags and enable the monitor coprocessor, native exception bits
180 cr0 |= (1 << 1);
181 cr0 &= ~(1 << 2);
182 cr0 &= ~(1 << 3);
183 cr0 |= (1 << 5);
184 asm volatile("mov %0, %%cr0" : : "r" (cr0));
185
186 // Enable the FPU
187 asm volatile("fninit");
188
189 // Check if SSE is supported
190 ASSERT(CPU::check_cpu_feature(CPU_FEATURE_EDX::SSE), "SSE not supported");
191
192 // Enable FSAVE, FSTORE and SSE instructions
193 cr4 |= (1 << 9);
194 cr4 |= (1 << 10);
195 asm volatile("mov %0, %%cr4" : : "r" (cr4));
196
197 // Check if XSAVE is supported
198 xsave_enabled = CPU::check_cpu_feature(CPU_FEATURE_ECX::XSAVE) && CPU::check_cpu_feature(CPU_FEATURE_ECX::OSXSAVE);
199 Logger::DEBUG() << "XSAVE: " << (xsave_enabled ? "Supported" : "Not Supported") << "\n";
200 if (!xsave_enabled) return;
201
202 // Enable the XSAVE and XRESTORE instructions
203 cr4 |= (1 << 18);
204 asm volatile("mov %0, %%cr4" : : "r" (cr4));
205
206 // Set the SSE and x87 bits
207 uint64_t xcr0;
208 asm volatile("xgetbv" : "=a" (xcr0) : "c" (0));
209 xcr0 |= 0x7;
210 asm volatile("xsetbv" : : "c" (0), "a" (xcr0));
211
212 // Check if AVX is supported
213 avx_enabled = CPU::check_cpu_feature(CPU_FEATURE_ECX::AVX);
214 Logger::DEBUG() << "AVX: " << (avx_enabled ? "Supported" : "Not Supported") << "\n";
215 if (!avx_enabled) return;
216
217 // Enable the AVX instructions
218 cr4 |= (1 << 14);
219 asm volatile("mov %0, %%cr4" : : "r" (cr4));
220
221 Logger::DEBUG() << "SSE Enabled\n";
222}

468 panic_lock.lock();

521cpu_status_t* CPU::prepare_for_panic(cpu_status_t* status, const string& msg) {
522 panic_lock.lock();
523
524 // If it may have occurred in a process, switch to the avoidable state
525 if (GlobalScheduler::system_scheduler() != nullptr && GlobalScheduler::current_process() != nullptr) {
526
527 // Get the current process
528 Process* process = GlobalScheduler::current_process();
529
530 // If the faulting address is in lower half just kill the process and move on
531 if (status && !memory::PhysicalMemoryManager::in_higher_region(status->rip)) {
532 Logger::Out() << "\n";
533 Logger::ERROR() << msg;
534 Logger::ERROR() << "CPU Panicked (i " << (int)status->interrupt_number << ") in process " << process->name.c_str() << " at 0x" << status->rip << " - killing process\n";
535 panic_lock.unlock();
536 return GlobalScheduler::force_remove_process(process);
537 }
538
539 // Otherwise occurred whilst the kernel was doing something for the process, thus a fault in the kernel fault
540 // and is unrecoverable
541 }
542
543 // Kill the others to make sure everything else stops
544 if (GlobalScheduler::is_active()) {
545 executing_core()->local_apic->send_interrupt(0, 0x81, EXCLUDE_SELF);
546 GlobalScheduler::deactivate();
547 }
548
549 // Prepare for printing debug messages
550 FramebufferConsole::print_logo(true);
551 Logger::active_logger() -> enable_all_log_writers();
552
553 // Unresumable state
554 panic_lock.unlock();
555 panic_core = executing_core();
556 return nullptr;
557}

564void CPU::find_cores() const {
565
566 // Now that memory is set up the vector can be used
567 cores.reserve(1);

581 index++;
582 }
