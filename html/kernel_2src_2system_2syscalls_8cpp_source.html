55cpu_status_t* SyscallManager::handle_interrupt(cpu_status_t* status) {
56
57 // Get the args from the cpu state
58 syscall_args_t args;
59 args.arg0 = status->rdi;
60 args.arg1 = status->rsi;
61 args.arg2 = status->rdx;
62 args.arg3 = status->r10;
63 args.arg4 = status->r8;
64 args.arg5 = status->r9;
65 args.return_value = 0;
66 args.return_state = status;
67
68 // Call the handler
69 uint64_t syscall = status->rax;
70 if (m_syscall_handlers[syscall] != nullptr)
71 args = *(m_syscall_handlers[syscall](&args));
72 else
73 Logger::ERROR() << "Syscall " << syscall << " not found\n";
74
75 // If there is a specific return state, use that
76 if (args.return_state != status)
77 return args.return_state;
78
79 // Update the cpu state
80 status->rdi = args.arg0;
81 status->rsi = args.arg1;
82 status->rdx = args.arg2;
83 status->r10 = args.arg3;
84 status->r8 = args.arg4;
85 status->r9 = args.arg5;
86 status->rax = args.return_value;
87
88 // Return the status
89 return status;
90}

120syscall_args_t* SyscallManager::syscall_klog(syscall_args_t* args) {
121
122 // Ensure a message was provided
123 char* message = (char*) args->arg0;
124 if(!message)
125 return args;
126
127 // If the first two characters are %h then no header
128 if (message[0] == '%' && message[1] == 'h')
129 Logger::Out() << message + 2;
130 else
131 Logger::INFO() << message;
132
133 return args;
134}
