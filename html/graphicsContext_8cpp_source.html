256uint32_t GraphicsContext::colour_to_int(const Colour& colour, uint32_t color_depth) {
257
258 switch(color_depth) {
259 case 8: {
260 uint32_t result = 0;
261 int mindistance = 0xfffffff;
262 for(uint32_t i = 0; i <= 255; ++i) {
263// Colour* c = &m_colour_pallet[i];
264 Colour* c = nullptr; //@todo fix this
265 int distance =
266 ((int) colour.red - (int) c->red) * ((int) colour.red - (int) c->red)
267 + ((int) colour.green - (int) c->green) * ((int) colour.green - (int) c->green)
268 + ((int) colour.blue - (int) c->blue) * ((int) colour.blue - (int) c->blue);
269 if(distance < mindistance) {
270 mindistance = distance;
271 result = i;
272 }
273 }
274 return result;
275 }
276 case 16: {
277 // 16-Bit colours RRRRRGGGGGGBBBBB
278 return ((uint16_t) (colour.red & 0xF8)) << 8
279 | ((uint16_t) (colour.green & 0xFC)) << 3
280 | ((uint16_t) (colour.blue & 0xF8) >> 3);
281 }
282 case 24: {
283 return (uint32_t) colour.red << 16
284 | (uint32_t) colour.green << 8
285 | (uint32_t) colour.blue;
286 }
287 default:
288 case 32: {
289 uint32_t red_hex = ((uint32_t) colour.red & 0xFF) << 16;
290 uint32_t green_hex = ((uint32_t) colour.green & 0xFF) << 8;
291 uint32_t blue_hex = (uint32_t) colour.blue & 0xFF;
292 uint32_t alpha_hex = ((uint32_t) colour.alpha & 0xFF) << 24;
293
294 uint32_t hex_value = red_hex | green_hex | blue_hex | alpha_hex;
295
296
297 return hex_value;
298 }
299 }

308Colour GraphicsContext::int_to_colour(uint32_t colour) {
309 switch(m_color_depth) {
310
311 case 8: {
312 // Return the colour from the palette
313 return m_colour_pallet[colour & 0xFF];
314 }
315
316 case 16: {
317 // 16-Bit Colour: 5 bits for red, 6 bits for green, 5 bits for blue (RRRRR,GGGGGG,BBBBB)
318 Colour result;
319
320 result.red = (colour & 0xF800) >> 8;
321 result.green = (colour & 0x07E0) >> 3;
322 result.blue = (colour & 0x001F) << 3;
323
324 return result;
325 }
326
327 case 24: {
328 // 24-Bit Colour: 8 bits for red, 8 bits for green, 8 bits for blue (RRRRRRRR,GGGGGGGG,BBBBBBBB)
329 Colour result;
330
331 result.red = (colour & 0xFF0000) >> 16;
332 result.green = (colour & 0x00FF00) >> 8;
333 result.blue = (colour & 0x0000FF);
334
335 return result;
336 }
337
338 default:
339 case 32: {
340 Colour result;
341
342 uint32_t hex_value = colour;
343 result.red = (hex_value >> 16) & 0xFF;
344 result.green = (hex_value >> 8) & 0xFF;
345 result.blue = hex_value & 0xFF;
346 result.alpha = (hex_value >> 24) & 0xFF;
347
348 return result;
349
350 }

474void GraphicsContext::draw_line(uint32_t x0, uint32_t y0, uint32_t x1, uint32_t y1, uint32_t colour) {
475
476 // Store the minimum and maximum y values
477 bool y_0_is_smaller = y0 < y1;
478 int32_t y_min = y_0_is_smaller ? y0 : y1;
479 int32_t y_max = y_0_is_smaller ? y1 : y0;
480
481 //Reverse the points to draw from left to right
482 if(x1 < x0) {
483 draw_line(x1, y1, x0, y0, colour);
484 return;
485 }
486
487 // Vertical line
488 if(x1 == x0) {
489 // Force the line to be within the screen
490 if(y_min < 0) y_min = 0;
491 if(y_max >= m_height)
492 y_max = m_height - 1;
493
494 // Mirror the Y axis as directly calling put_pixel will not do this
495 if(mirror_y_axis) {
496 int32_t temp = y_max;
497 y_max = m_height - y_min - 1;
498 y_min = m_height - temp - 1;
499 }
500
501 // Check that the line is within the screen
502 if(0 > x0 || x0 >= m_width) {
503 return;
504 }
505
506 // Draw the line
507 for(int32_t y = y_min; y <= y_max; ++y)
508 put_pixel(x0, y, colour);
509
510 return;
511 }
512
513 // Horizontal line
514 if(y1 == y0) {
515 // Ensure the line is within the screen
516 if(x0 < 0) x0 = 0;
517 if(x1 >= m_width) x1 = m_width - 1;
518
519 // Mirror the Y axis as directly calling put_pixel will not do this
520 if(mirror_y_axis)
521 y0 = m_height - y0 - 1;
522
523 // Check that the line is within the screen
524 if(0 > y0 || y0 >= m_height)
525 return;
526
527 // Draw the line
528 for(int32_t x = x0; x <= x1; ++x)
529 put_pixel(x, y0, colour);
530 }
531
532 // If the line is not horizontal or vertical then it must be a diagonal line
533 // Find the slope of the line
534// float slope = ((float) (y1 - y0)) / (x1 - x0);
535 int slope = 1; // replace when in userland
536
537 // A slope that is more horizontal should be drawn by incrementing x
538 if(-1 <= slope && slope <= 1) {
539 int y = y0;
540 for(int32_t x = x0; x <= x1; x++, y += slope)
541 put_pixel(x, (int32_t) y, colour);
542 }
543
544 // A slope that is more vertical should be drawn by incrementing y
545 else {
546 // Invert the slope
547 slope = 1 / slope;
548
549 int x = x0;
550 for(int32_t y = y_min; y <= y_max; x += slope, y++)
551 put_pixel((int32_t) x, y, colour);
552 }
553}

672
673 // Make sure the circle is with in the width and height of the screen
674 if(x0 < 0) x0 = 0;

681 y0 = m_height - y0 - 1;
682
683
684 // Begin drawing at the left most point of the circle and draw a line to the right most point of the circle
685 for(int32_t x = -radius; x <= radius; ++x) {
686

688 for(int32_t y = -radius; y <= radius; ++y) {
689

691 if(x * x + y * y <= radius * radius && x * x + y * y >= (radius - 1) * (radius - 1))
