53size_t BridgeHandler::store_data(const void *buffer, size_t size, bool reserve_only) {
54
55 // Nothing to store
56 if (!size)
57 return 0;
58
59 bool is_mapped = size >= MAX_COPY_SIZE;
60
61 // Set up the storage
62 void* data;
63 if (is_mapped) {
64 MemoryManager* caller = GlobalScheduler::current_process()->memory_manager;
65 MemoryManager::switch_active_memory_manager(m_owner_process->memory_manager);
66 data = m_owner_process->memory_manager->vmm()->load_range_from_process(caller->vmm(), buffer, size);
67 MemoryManager::switch_active_memory_manager(caller);
68 } else {
69 m_lock.lock();
70 data = allocate(size);
71 m_lock.unlock();
72 }
73 ASSERT(data != nullptr, "Could not allocate memory for bridge\n");
74
75 // Copy the data
76 if (!reserve_only && !is_mapped)
77 memcpy(data, buffer, size);
78
79 // If not mapped then its an offset into the process's own virtual address for the shared mem
80 return (size_t)data - (is_mapped ? 0 : m_data_region);
81
82}

89service_resource_message_t* BridgeHandler::aquire_slot() {
90
91 // Find a spot
92 while (true) {
93
94 // Get the data
95 size_t head = __atomic_load_n(&m_message_ring->head, __ATOMIC_RELAXED);
96 size_t tail = __atomic_load_n(&m_message_ring->tail, __ATOMIC_ACQUIRE);
97
98 ASSERT((head < MESSAGE_SLOTS && tail < MESSAGE_SLOTS), "invalid queue indexes for resource bridge");
99
100 // No available spots to claim
101 size_t next = (head + 1) % MESSAGE_SLOTS;
102 if (next == tail) {
103 GlobalScheduler::current_thread()->yield();
104 continue;
105 }
106
107 // Try to claim a spot
108 if(!__atomic_compare_exchange_n(&m_message_ring->head, &head, next,true,__ATOMIC_ACQ_REL,__ATOMIC_RELAXED))
109 continue;
110
111 // Claim slot
112 size_t slot = head;
113 return &m_message_ring->ring_buffer[slot];
114 }
115}

139int64_t BridgeHandler::send_to_bridge(size_t id, ServiceResourceCommand command, const void *buffer, size_t size, size_t flags) {
140
141 // Copy data into shared region
142 bool is_read = command == ServiceResourceCommand::R_READ; // || command == BridgeResourceCommand::R_READ_ATTR;
143 bool is_page_mapped = size >= MAX_COPY_SIZE;
144 size_t offset = store_data(buffer, size, is_read);
145
146 // Construct the message
147 auto slot = aquire_slot();
148 __atomic_store_n(&slot->sending_pid, GlobalScheduler::current_process()->pid(), __ATOMIC_RELAXED);
149 __atomic_store_n(&slot->resource_id, id, __ATOMIC_RELAXED);
150 __atomic_store_n(&slot->flags, flags, __ATOMIC_RELAXED);
151 __atomic_store_n(&slot->command, (uint8_t)command, __ATOMIC_RELAXED);
152 __atomic_store_n(&slot->data_size, size, __ATOMIC_RELAXED);
153 __atomic_store_n(&slot->data_offset, offset, __ATOMIC_RELAXED);
154 __atomic_store_n(&slot->data_page_mapped, is_page_mapped, __ATOMIC_RELAXED);
155 __atomic_store_n(&slot->response, 0, __ATOMIC_RELAXED);
156 __atomic_store_n(&slot->state, (uint8_t)ServiceMessageSlotState::REQUEST, __ATOMIC_RELEASE);
157
158 // Wait for a response
159 while (__atomic_load_n(&slot->state, __ATOMIC_ACQUIRE) == (uint8_t)ServiceMessageSlotState::REQUEST)
160 GlobalScheduler::current_thread()->yield();
161
162 // Read request
163 if (is_read && !is_page_mapped)
164 memcpy((void*)buffer, (void*)m_data_region + offset, size);
165
166 // Free resources
167 if (size != 0) {
168 if (is_page_mapped) {
169 MemoryManager* caller = GlobalScheduler::current_process()->memory_manager;
170 MemoryManager::switch_active_memory_manager(m_owner_process->memory_manager);
171 m_owner_process->memory_manager->vmm()->unload_range_from_process((void*)offset, size);
172 MemoryManager::switch_active_memory_manager(caller);
173 }
174 else {
175 m_lock.lock();
176 unallocate((void*)m_data_region + offset);
177 m_lock.unlock();
178 }
179 }
180
181 // Get the response and then release the slot
182 int64_t response = __atomic_load_n(&slot->response, __ATOMIC_RELAXED);
183 __atomic_store_n(&slot->state, (uint8_t)ServiceMessageSlotState::FREE,__ATOMIC_RELEASE);
184
185 return response;
186}

279Resource* BridgeResourceRegistry::get_resource(const string &name) {
280
281 // Resource already opened
282 auto resource = BaseResourceRegistry::get_resource(name);
283 if(resource != nullptr)
284 return resource;
285
286 // Setup a temporary resource
287 auto attempt_resource = new BridgeResource(name, 0, m_type);
288 attempt_resource->attach_wrapper(&m_handler);
289
290 // Try open on the server
291 if (attempt_resource->get_on_bridge()) {
292 register_resource(attempt_resource);
293 return attempt_resource;
294 }
295
296 // Must have failed to get on server

298 return nullptr;
