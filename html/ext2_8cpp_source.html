26Ext2Volume::Ext2Volume(Disk* disk, lba_t partition_offset)
27: disk(disk),
28 partition_offset(partition_offset),
29 superblock({})
30{
31
32 // Read superblock
33 buffer_t superblock_buffer(&superblock, 1024);
34 disk->read(partition_offset + 2, &superblock_buffer, 512);
35 disk->read(partition_offset + 3, &superblock_buffer, 512);
36
37 // Validate signature
38 ASSERT(superblock.signature == 0xEF53, "Ext2 Filesystem doesnt have a valid signature\n");
39
40 // Version 0 has constant inode info
41 if(superblock.version_major < 1) {
42 superblock.first_inode = 11;
43 superblock.inode_size = 128;
44 }
45
46 // Parse the superblock
47 block_size = 1024 << superblock.block_size;
48 total_block_groups = (superblock.total_blocks + superblock.blocks_per_group - 1) / superblock.blocks_per_group;
49 block_group_descriptor_table_block = superblock.starting_block + 1;
50 block_group_descriptor_table_size = total_block_groups * sizeof(block_group_descriptor_t);
51 pointers_per_block = block_size / sizeof(uint32_t);
52 inodes_per_block = block_size / superblock.inode_size;
53 sectors_per_block = block_size / 512;
54
55 // Read the block groups
56 block_groups = new block_group_descriptor_t* [total_block_groups] { nullptr };
57 uint32_t bgdt_lba = partition_offset + block_group_descriptor_table_block * sectors_per_block;
58 uint32_t sectors_to_read = (block_group_descriptor_table_size + block_size - 1) / block_size * sectors_per_block;
59 buffer_t bg_buffer(sectors_to_read * 512);
60 for(uint32_t i = 0; i < sectors_to_read; ++i)
61 disk->read(bgdt_lba + i, &bg_buffer, 512);
62
63 // Cache the groups
64 bgdt_cache = new uint8_t[sectors_to_read * 512];
65 memcpy(bgdt_cache, bg_buffer.raw(), sectors_to_read * 512);
66
67 // Store the block groups
68 for(uint32_t i = 0; i < total_block_groups; ++i) {
69 block_groups[i] = new block_group_descriptor_t;
70 memcpy(block_groups[i], bg_buffer.raw() + i * sizeof(block_group_descriptor_t),
71 sizeof(block_group_descriptor_t));
72 }
73}

191Vector<uint32_t> Ext2Volume::allocate_blocks(uint32_t amount) {
192
193 // No blocks to allocate
194 if(!amount)
195 return { 1, 0 };
196
197 // Find the block group with enough free blocks
198 block_group_descriptor_t* block_group = block_groups[0];
199 for(uint32_t bg_index = 0; bg_index < total_block_groups; block_group = block_groups[++bg_index])
200 if(block_group->free_blocks >= amount)
201 return allocate_group_blocks(bg_index, amount);
202
203 // No block group can contain the block so split across multiple
204 Vector<uint32_t> result { };
205 while(amount > 0) {
206
207 // Find the block group with most free blocks
208 block_group = block_groups[0];
209 uint32_t bg_index = 0;
210 for(; bg_index < total_block_groups; ++bg_index)
211 if(block_groups[bg_index]->free_blocks > block_group->free_blocks)
212 block_group = block_groups[bg_index];
213
214 // No space
215 if(block_group->free_blocks == 0)
216 return { 1, 0 };
217
218 // Allocate the remaining blocks
219 auto allocated = allocate_group_blocks(bg_index, 1);
220 amount -= allocated.size();
221 for(auto block : allocated)
222 result.push_back(block);
223 }
224
225 return result;
226}

235common::Vector<uint32_t> Ext2Volume::allocate_group_blocks(uint32_t block_group, uint32_t amount) {
236
237 // Ensure enough space
238 block_group_descriptor_t* descriptor = block_groups[block_group];
239 if(amount > descriptor->free_blocks)
240 return { 1, 0 };
241
242 // Prepare
243 Vector<uint32_t> result { };
244 buffer_t zeros(block_size);
245 zeros.clear();
246
247 // Read bitmap
248 buffer_t bitmap(block_size);
249 read_block(descriptor->block_usage_bitmap, &bitmap);
250
251 // Allocate the blocks
252 for(uint32_t i = 0; i < superblock.blocks_per_group; ++i) {
253
254 // Block is already used
255 if((bitmap.raw()[i / 8] & (1u << (i % 8))) != 0)
256 continue;
257
258 // Mark as used
259 descriptor->free_blocks--;
260 superblock.unallocated_blocks--;
261 bitmap.raw()[i / 8] |= (uint8_t) (1u << (i % 8));
262
263 // Zero out data
264 uint32_t block = block_group * superblock.blocks_per_group + superblock.starting_block + i;
265 write_block(block, &zeros);
266 result.push_back(block);
267
268 // All done
269 amount--;
270 if(!amount)
271 break;
272 }
273
274 // Save the changed metadata
275 write_block(descriptor->block_usage_bitmap, &bitmap);
276 write_back_block_groups();
277 write_back_superblock();
278
279 return result;
280}

290void Ext2Volume::free_group_blocks(uint32_t block_group, uint32_t amount, uint32_t start) {
291
292 // Read bitmap
293 block_group_descriptor_t* descriptor = block_groups[block_group];
294 buffer_t bitmap(block_size);
295 read_block(descriptor->block_usage_bitmap, &bitmap);
296
297 // Convert start to be index based on the group instead of global
298 start -= (block_group * superblock.blocks_per_group + superblock.starting_block);
299
300 // Free the blocks
301 for(uint32_t i = start; i < start + amount; ++i) {
302
303 // Block is already free (shouldn't happen)
304 if((bitmap.raw()[i / 8] & (1u << (i % 8))) == 0)
305 continue;
306
307 // Mark as free
308 descriptor->free_blocks++;
309 superblock.unallocated_blocks++;
310 bitmap.raw()[i / 8] &= ~(1u << (i % 8));
311 }
312
313 // Save the changed metadata
314 write_block(descriptor->block_usage_bitmap, &bitmap);
315 write_back_block_groups();
316 write_back_superblock();
317
318}

324void Ext2Volume::write_back_block_groups() const {
325
326 // Locate the block groups
327 uint32_t bgdt_blocks = (block_group_descriptor_table_size + block_size - 1) / block_size;
328 uint32_t sectors_to_write = bgdt_blocks * sectors_per_block;
329 uint32_t bgdt_lba = partition_offset + block_group_descriptor_table_block * sectors_per_block;
330
331 // Copy the block groups into the buffer
332 buffer_t bg_buffer(sectors_to_write * 512);
333 bg_buffer.clear();
334 for(uint32_t i = 0; i < total_block_groups; ++i)
335 bg_buffer.copy_from(block_groups[i], sizeof(block_group_descriptor_t));
336
337 // Write the buffer to disk
338 for(uint32_t i = 0; i < sectors_to_write; ++i) {
339
340 // Group didnt change
341 if(bgdt_cache && memcmp(bgdt_cache + i * 512, bg_buffer.raw() + i * 512, 512) == 0)
342 continue;
343
344 // Write to disk
345 bg_buffer.set_offset(i * 512);
346 disk->write(bgdt_lba + i, &bg_buffer, 512);
347
348 // Write to cache
349 if(bgdt_cache)
350 memcpy(bgdt_cache + i * 512, bg_buffer.raw() + i * 512, 512);
351 }
352}

385uint32_t Ext2Volume::create_inode(bool is_directory) {
386
387 ext2_lock.lock();
388
389 // Find the block group with enough free inodes
390 block_group_descriptor_t* block_group = block_groups[0];
391 uint32_t bg_index = 0;
392 for(; bg_index < total_block_groups; block_group = block_groups[++bg_index])
393 if(block_group->free_inodes >= 1)
394 break;
395
396 // Read bitmap
397 buffer_t bitmap(block_size);
398 read_block(block_group->block_inode_bitmap, &bitmap);
399
400 // First group contains reserved inodes
401 uint32_t inode_index = 0;
402 if(bg_index == 0 && superblock.first_inode > 1)
403 inode_index = superblock.first_inode - 1;
404
405 // Find a free inode
406 for(; inode_index < superblock.inodes_per_group; ++inode_index) {
407
408 // Block is already used
409 if((bitmap.raw()[inode_index / 8] & (1u << (inode_index % 8))) != 0)
410 continue;
411
412 // Mark as used
413 block_group->free_inodes--;
414 superblock.unallocated_inodes--;
415 bitmap.raw()[inode_index / 8] |= (uint8_t) (1u << (inode_index % 8));
416
417 break;
418 }
419
420 // Convert into the 1-based inode index in the group
421 inode_index += bg_index * superblock.inodes_per_group + 1;
422
423 // Save the changed metadata
424 write_block(block_group->block_inode_bitmap, &bitmap);
425 write_back_block_groups();
426 write_back_superblock();
427
428 // Create the inode
429 inode_t inode { };
430 //TODO: CLOCK inode.creation_time = time_to_epoch(Clock::active_clock()->get_time());
431 //TODO: CLOCK inode.last_modification_time = time_to_epoch(Clock::active_clock()->get_time());
432 inode.block_pointers[0] = allocate_block();
433 inode.hard_links = is_directory ? 2 : 1;
434 inode.type = ((uint16_t) (is_directory ? InodeType::DIRECTORY : InodeType::FILE) >> 12) & 0xF;
435 inode.permissions =
436 (uint16_t) (is_directory ? InodePermissionsDefaults::DIRECTORY : InodePermissionsDefaults::FILE) & 0x0FFF;
437 write_inode(inode_index, &inode);
438
439 ext2_lock.unlock();
440 return inode_index;
441}

449void Ext2Volume::free_inode(uint32_t inode) {
450
451 // Find the block group containing the inode
452 uint32_t bg_index = (inode - 1) / superblock.inodes_per_group;
453 block_group_descriptor_t* block_group = block_groups[bg_index];
454
455 // Read bitmap
456 buffer_t bitmap(block_size);
457 read_block(block_group->block_inode_bitmap, &bitmap);
458
459 // First group contains reserved inodes
460 uint32_t inode_index = (inode - 1) % superblock.inodes_per_group;
461 if(bg_index == 0 && (inode_index < (superblock.first_inode - 1)))
462 return;
463
464 // Mark as used
465 block_group->free_inodes++;
466 superblock.unallocated_inodes++;
467 bitmap.raw()[inode_index / 8] &= (uint8_t) ~(1u << (inode_index % 8));
468
469 // Save the changed metadata
470 write_block(block_group->block_inode_bitmap, &bitmap);
471 write_back_block_groups();
472 write_back_superblock();

479void Ext2Volume::free_blocks(const common::Vector<uint32_t>& blocks) {
480
481 // No blocks to free
482 if(blocks.empty())
483 return;
484
485 uint32_t start = blocks[0];
486 uint32_t previous = start;
487 uint32_t amount = 1;
488
489 // Free each adjacent set of blocks
490 for(auto& block : blocks) {
491
492 // First is already accounted for
493 if(block == start)
494 continue;
495
496 // Is this block adjacent
497 if((previous + 1) == block) {
498 previous = block;
499 amount += 1;
500 continue;
501 }
502
503 // Adjacent set has ended
504 uint32_t group = (start - superblock.starting_block) / superblock.blocks_per_group;
505 free_group_blocks(group, amount, start);
506
507 // Reset
508 start = block;
509 previous = start;
510 amount = 1;
511 }
512
513 // Account for the last set of blocks in the loop
514 uint32_t group = (start - superblock.starting_block) / superblock.blocks_per_group;
515 free_group_blocks(group, amount, start);
516}

566void InodeHandler::parse_indirect(uint32_t level, uint32_t block, buffer_t* buffer) {
567
568 // Invalid
569 if(block == 0)
570 return;
571
572 // Read the block
573 m_volume->read_block(block, buffer);
574 auto* pointers = (uint32_t*) (buffer->raw());
575
576 // Parse the pointers
577 for(size_t i = 0; i < m_volume->pointers_per_block; ++i) {
578 uint32_t pointer = pointers[i];
579
580 // Invaild
581 if(pointer == 0)
582 break;
583
584 // Has indirect sub entries
585 if(level > 1) {
586 parse_indirect(level - 1, pointer, buffer);
587 continue;
588 }
589
590 // Parse the entry
591 block_cache.push_back(pointer);
592 }
593}

603void InodeHandler::write_indirect(uint32_t level, uint32_t& block, size_t& index) {
604
605 // Nothing left to write
606 size_t remaining = block_cache.size() - index;
607 if(remaining == 0 || index >= block_cache.size())
608 return;
609
610 // Level hasn't been set yet
611 if(block == 0)
612 block = m_volume->allocate_block();
613
614 // Allocate a local buffer for this recursion level
615 buffer_t buffer(m_volume->block_size);
616 buffer.clear();
617 auto* pointers = (uint32_t*) buffer.raw();
618
619 // Write the pointers
620 for(size_t i = 0; i < m_volume->pointers_per_block; ++i) {
621
622 // Invalid
623 if(index >= block_cache.size())
624 break;
625
626 // Has indirect
627 if(level > 1) {
628 write_indirect(level - 1, pointers[i], index);
629 continue;
630 }
631
632 // Save the pointer
633 pointers[i] = block_cache[index++];
634 }
635
636 m_volume->write_block(block, &buffer);
637}

644void InodeHandler::store_blocks(Vector<uint32_t> const& blocks) {
645
646 // Store in cache
647 for(auto block : blocks)
648 block_cache.push_back(block);
649
650 // Direct blocks
651 for(uint32_t i = 0; i < 12; ++i)
652 inode.block_pointers[i] = i < block_cache.size() ? block_cache[i] : 0;
653
654 // No need to do any indirects
655 if(block_cache.size() < 12)
656 return;
657
658 // Setup Recursive blocks
659 size_t index = 12;
660
661 // Write the blocks
662 uint32_t indirect_blocks[3] = { inode.l1_indirect, inode.l2_indirect, inode.l3_indirect };
663 for(int i = 0; i < 3; ++i) {
664
665 // Have to use temp because of packed field
666 uint32_t temp = indirect_blocks[i];
667 write_indirect(i + 1, temp, index);
668 indirect_blocks[i] = temp;
669 }
670
671 // Save the new blocks
672 inode.l1_indirect = indirect_blocks[0];
673 inode.l2_indirect = indirect_blocks[1];
674 inode.l3_indirect = indirect_blocks[2];
675
676 // NOTE: Blocks get allocated when writing indirects. This is then saved later in the write() function
677}

755void Ext2File::write(buffer_t* data, size_t amount) {
756
757 // Nothing to write
758 if(amount == 0)
759 return;
760
761 // Prepare for writing
762 m_volume->ext2_lock.lock();
763 const uint32_t block_size = m_volume->block_size;
764 buffer_t buffer(block_size);
765
766 // Expand the file
767 if(m_offset + amount > m_size)
768 m_size = m_inode.grow((m_offset + amount) - m_size, false);
769
770 // Save the updated metadata
771 //TODO: CLOCK m_inode.inode.last_modification_time = time_to_epoch(Clock::active_clock()->get_time());
772 m_inode.save();
773
774 // Convert bytes to blocks
775 uint32_t block_start = m_offset / block_size;
776 uint32_t block_offset = m_offset % block_size;
777
778 // Write each block
779 size_t current_block = block_start;
780 size_t written = 0;
781 while(written < amount) {
782
783 uint32_t block = m_inode.block_cache[current_block++];
784
785 // Where in this block to start writing
786 size_t buffer_start = (current_block - 1 == block_start) ? block_offset : 0;
787 size_t writable = (amount - written < block_size - buffer_start) ? (amount - written) : (block_size - buffer_start);
788
789 // Only read the old contents back when part of the block is kept
790 if (buffer_start != 0 || writable != block_size)
791 m_volume->read_block(block, &buffer);
792
793 // Update the block
794 buffer.copy_from(data, writable, buffer_start, written);
795 m_volume->write_block(block, &buffer);
796 written += writable;
797 }
798
799 // Clean up
800 m_offset += amount;
801 m_volume->ext2_lock.unlock();
802}

810void Ext2File::read(buffer_t* data, size_t amount) {
811
812 // Nothing to read
813 if(m_size == 0 || amount == 0)
814 return;
815
816 // Prepare for reading
817 m_volume->ext2_lock.lock();
818 const uint32_t block_size = m_volume->block_size;
819 buffer_t buffer(block_size);
820
821 // Force bounds
822 if(m_offset + amount > m_size)
823 amount = m_size - m_offset;
824
825 // Convert bytes to blocks
826 uint32_t block_start = m_offset / block_size;
827 uint32_t block_offset = m_offset % block_size;
828
829 // Read each block
830 size_t current_block = block_start;
831 size_t read = 0;
832 while(read < amount) {
833
834 // Read the block
835 uint32_t block = m_inode.block_cache[current_block++];
836 m_volume->read_block(block, &buffer);
837
838 // Where in this block to start reading
839 size_t buffer_start = (current_block - 1 == block_start) ? block_offset : 0;
840 size_t readable = (amount - read < block_size - buffer_start) ? (amount - read) : (block_size - buffer_start);
841
842 // Read the block
843 buffer.copy_to(data, readable, buffer_start, read);
844 read += readable;
845
846 }
847
848 // Clean up
849 m_offset += amount;
850 m_volume->ext2_lock.unlock();
851}

878void Ext2Directory::parse_block(buffer_t* buffer) {
879
880 size_t offset = 0;
881 while(offset < m_volume->block_size) {
882
883 // Read the entry
884 auto* entry = (directory_entry_t*) (buffer->raw() + offset);
885 m_entries.push_back(*entry);
886
887 // Not valid
888 if(entry->inode == 0 || entry->name_length == 0)
889 break;
890
891 // Parse
892 string filename(buffer->raw() + offset + sizeof(directory_entry_t), entry->name_length);
893 m_entry_names.push_back(filename);
894 uint32_t inode = entry->inode;
895
896 // Create the object
897 switch((EntryType) entry->type) {
898
899 case EntryType::FILE:
900 m_files.push_back(new Ext2File(m_volume, inode, filename));
901 break;

905 break;
906
907 default:

909
910 }

912 // Go to next
913 offset += entry->size;
914 }

924void Ext2Directory::remove_entry(string const& name, bool is_directory, bool clear) {
925
926 // Find the entry
927 uint32_t index = 0;
928 directory_entry_t* entry = nullptr;
929 for(; index < m_entries.size(); ++index)
930 if(m_entry_names[index] == name) {
931 entry = &m_entries[index];
932 break;
933 }
934
935 // No entry found
936 if(!entry || entry->type != (uint8_t) (is_directory ? EntryType::DIRECTORY : EntryType::FILE))
937 return;
938
939 // Clear the inode
940 if(clear) {
941 InodeHandler inode(m_volume, entry->inode);
942 inode.free();
943 }
944
945 // Remove the reference from this directory
946 m_entries.erase(entry);
947 m_entry_names.erase(m_entry_names.begin() + index);
948 write_entries();

1008void Ext2Directory::write_entries() {
1009
1010 // Calculate the size needed to store the entries and the null entry
1011 size_t size_required = sizeof(directory_entry_t);
1012 for(uint32_t i = 0; i < m_entries.size(); ++i) {
1013 size_t size = sizeof(directory_entry_t) + m_entry_names[i].length() + 1;
1014 size += (size % 4) ? 4 - size % 4 : 0;
1015 size_required += size;
1016 }
1017
1018 // Expand the directory
1019 size_t blocks_required = m_volume->bytes_to_blocks(size_required);
1020 if(blocks_required > m_inode.block_cache.size())
1021 m_inode.grow((blocks_required - m_inode.block_cache.size()) * m_volume->block_size, false);
1022
1023 // Prepare for writing
1024 m_volume->ext2_lock.lock();
1025 const uint32_t block_size = m_volume->block_size;
1026 buffer_t buffer(block_size, false);
1027 buffer.clear();
1028
1029 // Save the updated metadata
1030 m_inode.set_size(blocks_required * block_size);
1031 //TODO: CLOCK m_inode.inode.last_modification_time = time_to_epoch(Clock::active_clock()->get_time());
1032 m_inode.save();
1033
1034 // Write each entry
1035 size_t current_block = 0;
1036 size_t buffer_offset = 0;
1037 for(uint32_t i = 0; i < m_entries.size(); ++i) {
1038
1039 // Get the current entry
1040 directory_entry_t& entry = m_entries[i];
1041 char* name = m_entry_names[i].c_str();
1042
1043 // Update the size
1044 entry.name_length = m_entry_names[i].length();
1045 entry.size = sizeof(directory_entry_t) + entry.name_length + 1;
1046 entry.size += (entry.size % 4) ? 4 - (entry.size % 4) : 0;
1047
1048 // Entry needs to be stored in the next block
1049 if(entry.size + buffer_offset > block_size) {
1050 m_volume->write_block(m_inode.block_cache[current_block], &buffer);
1051 buffer.clear();
1052 current_block++;
1053 buffer_offset = 0;
1054 }
1055
1056 // If it is the last entry it takes up the rest of the block
1057 if(i == m_entries.size() - 1)
1058 entry.size = block_size - buffer_offset;
1059
1060 // Copy the entry and the name
1061 buffer.copy_from(&entry, sizeof(entry), buffer_offset);
1062 buffer.copy_from(name, entry.name_length + 1, buffer_offset + sizeof(entry));
1063 buffer_offset += entry.size;
1064 }
1065
1066 // Save the last block
1067 m_volume->write_block(m_inode.block_cache[current_block], &buffer);
1068
1069 // Clean up
1070 m_volume->ext2_lock.unlock();
1071}
