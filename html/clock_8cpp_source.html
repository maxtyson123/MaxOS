137void Clock::setup_apic_clock(hardwarecommunication::LocalAPIC* local_apic) const {

156common::time_t Clock::get_time() {

162 // Read the time from the clock
163 time_t time{};

166 time.day = binary_representation(read_hardware_clock(0x7));
167 time.hour = binary_representation(read_hardware_clock(0x4));
168 time.minute = binary_representation(read_hardware_clock(0x2));
169 time.second = binary_representation(read_hardware_clock(0x0));
170
171 // If the clock is using 12hr format and PM is set then add 12 to the hour
172 if (!m_24_hour_clock && (time.hour & 0x80) != 0)
173 time.hour = ((time.hour & 0x7F) + 12) % 24;
174
175 return time;

214uint32_t PIT::ticks_per_ms() {
215
216 // Set the redirect for the timer interrupt
217 interrupt_redirect_t redirect = {
218 .type = 0x2,
219 .index = 0x14,
220 .interrupt = 0x22,
221 .destination = 0x00,
222 .flags = 0x00,
223 .mask = true,
224 };
225 m_io_apic->set_redirect(&redirect);
226
227 // Configure the PIT clock
228 pit_command_t command = {
229 .bcd_mode = (uint8_t) BCDMode::BINARY,
230 .operating_mode = (uint8_t) OperatingMode::RATE_GENERATOR,
231 .access_mode = (uint8_t) AccessMode::LOW_HIGH_BYTE,
232 .channel = (uint8_t) Channel::INTERRUPT
233 };
234 m_command_port.write(*(uint8_t *) &command);
235
236 // Set the clock rate to 1 ms;
237 uint16_t rate = 1193182 / 1000;
238 m_data_port.write(rate & 0xFF);
239 m_data_port.write(rate >> 8);
240
241 // Stop the clock
242 m_local_apic->write(0x380, 0x00);
243
244 // Set the divisor to 2
245 m_local_apic->write(0x3E0, 0x0);
246
247 // Unmask the PIT interrupt
248 m_io_apic->set_redirect_mask(redirect.index, false);
249
250 // Calculate the number of ticks in 1 ms
251 auto max = (uint32_t) -1;
252 m_local_apic->write(0x380, max);
253
254 while (m_ticks < TICK_CALIBRATE_LENGTH)
255 asm volatile("nop");
256
257 uint32_t elapsed = max - (m_local_apic->read(0x390));
258 uint32_t ticks_per_ms = elapsed / TICK_CALIBRATE_LENGTH;
259
260 Logger::DEBUG() << "Ticks per ms: " << (int) ticks_per_ms << "\n";
261
262 // Disable the PIT interrupt again
263 m_local_apic->write(0x380, 0x00);
264 m_io_apic->set_redirect_mask(redirect.index, true);
265
266 return ticks_per_ms;
267}
