150void GlobalScheduler::load_multiboot_elfs(Multiboot* multiboot) {
151
152 for (multiboot_tag* tag = multiboot->start_tag(); tag->type != MULTIBOOT_TAG_TYPE_END; tag = (struct multiboot_tag*) ((multiboot_uint8_t*) tag + ((tag->size + 7) & ~7))) {
153
154 // Tag is not an module
155 if (tag->type != MULTIBOOT_TAG_TYPE_MODULE)
156 continue;
157
158 // Get the tag
159 auto* module = (struct multiboot_tag_module*) tag;
160 auto start = (uintptr_t)PhysicalMemoryManager::to_dm_region(module->mod_start);
161 Logger::DEBUG() << "Creating process from multiboot module for " << module->cmdline << " (at 0x" << (uint64_t) module->mod_start << ")\n";
162
163 // Handle the init ramdisk
164 if (strcmp(module->cmdline, "initrd") == 0)
165 prepare_initrd(module);
166
167 // Try to create the elf from the module
168 ELF64 elf(start);
169 if (!elf.is_valid())
170 continue;
171
172 // Create an array of args for the process
173 char* args[1] = {module->cmdline};
174
175 // Create the process
176 auto* process = new Process(module->cmdline, args, 1, &elf);
177 system_scheduler() -> add_process(process);
178
179 Logger::DEBUG() << "ELF loaded to pid " << process->pid() << "\n";
180 }

190void GlobalScheduler::prepare_initrd(multiboot_tag_module* module) {
191
192 // Find the Init process
193 Process* init_process = get_process(4);
194 ASSERT(init_process != nullptr, "Can't find init process\n");
195
196 // Extract info
197 auto start = (uintptr_t)PhysicalMemoryManager::to_dm_region(module->mod_start);
198 uint32_t size = module->mod_end - module->mod_start;
199
200 // Prepare
201 size_t actual_size;
202 MemoryManager::switch_active_memory_manager(init_process->memory_manager);
203 init_process->memory_manager->vmm()->allocate(0xFFF000, size + sizeof(uint32_t), actual_size, WRITE | PRESENT);
204
205 // Copy
206 memcpy((void*)0xFFF000, &size, sizeof(uint32_t));
207 memcpy((void*)(0xFFF000 + sizeof(uint32_t)), (void*)start, size);
208 MemoryManager::switch_active_memory_manager(MemoryManager::s_kernel_memory_manager);
209}

446cpu_status_t* Scheduler::schedule(cpu_status_t* cpu_state) {
447
448 // Scheduler cant schedule anything
449 if (m_threads.empty() || !m_active)
450 return cpu_state;
451
452 // Thread that we are dealing with
453 Thread* current_thread = m_threads[m_next_thread_index];
454
455 // Ticked
456 m_ticks++;
457 current_thread->ticks++;
458
459 // The idle thread has no time slice to protect
460 Process* running_process = current_process();
461 if (running_process && running_process->is_kernel)
462 return schedule_next(cpu_state);
463
464 // Thread hasn't used its time slot yet
465 if(current_thread->ticks % TICKS_PER_EVENT)
466 return cpu_state;
467
468 // Schedule the next thread
469 return schedule_next(cpu_state);
470}

480cpu_status_t* Scheduler::schedule_next(cpu_status_t* cpu_state) {
481
482 // Get the current state
483 Thread* current_thread = m_threads[m_next_thread_index];
484 Process* owner_process = current_process();
485 auto old_tid = m_next_thread_index;
486
487 // Save the executing thread state
488 current_thread->execution_state = *cpu_state;
489 current_thread->save_sse_state();
490 if (current_thread->thread_state == ThreadState::RUNNING)
491 current_thread->thread_state = ThreadState::READY;
492
493 // Find a free thread to run
494 while ((++m_next_thread_index) != old_tid){
495 m_next_thread_index %= m_threads.size();
496
497 // Get the current thread
498 current_thread = m_threads[m_next_thread_index];
499 owner_process = current_process();
500
501 // Handle state changes
502 switch (current_thread->thread_state) {
503
504 case ThreadState::SLEEPING:
505
506 // If the wake-up time hasn't occurred yet, run the next thread
507 if (current_thread->wakeup_time > TICKS_PER_EVENT){
508 current_thread->wakeup_time -= TICKS_PER_EVENT;
509 continue;
510 }
511
512 break;
513
514 case ThreadState::STOPPED:
515
516 // Find the process that has the thread and remove it
517 for (auto thread: owner_process->threads()) {
518 if (thread == current_thread) {
519 owner_process->remove_thread(thread->tid);
520 break;
521 }
522 }
523
524 // Remove the thread
525 m_threads.erase(m_threads.begin() + m_next_thread_index);
526 if(owner_process->threads().empty())
527 remove_process(owner_process);
528 continue;
529
530 case ThreadState::WAITING:
531 continue;
532
533 default:
534 break;
535 }
536 break;
537 }
538
539 int pid0 = m_processes.size() > 0 ? (int) m_processes[0]->pid() : -1;
540 int pid1 = m_processes.size() > 1 ? (int) m_processes[1]->pid() : -1;
541 int pid2 = m_processes.size() > 2 ? (int) m_processes[2]->pid() : -1;
542 ASSERT(owner_process != nullptr, "No owner process: tid %d parent_pid %d core %d index %d threads %d processes %d pids %d %d %d\n",
543 (int) current_thread->tid, (int) current_thread->parent_pid, (int) m_core_id,
544 (int) m_next_thread_index, (int) m_threads.size(), (int) m_processes.size(), pid0, pid1, pid2);
545
546
547 // Load the thread's state
548 return load_process(owner_process, current_thread);
549}

558cpu_status_t* Scheduler::load_process(Process* process, Thread* thread) {
559
560 // Prepare the next thread to run
561 thread->thread_state = ThreadState::RUNNING;
562 thread->restore_sse_state();
563
564 // Load into the threads memory region
565 MemoryManager::switch_active_memory_manager(process->memory_manager);
566

652uint64_t Scheduler::remove_process(Process* process) {
653
654 // Check if the process has no threads
655 if (!process->threads().empty()) {
656
657 // Set the threads to stopped or remove them if forced
658 for (auto thread: process->threads())
659 thread->thread_state = ThreadState::STOPPED;
660
661 // Need to wait until the threads are stopped before removing the process (this will be called again when all threads are stopped)
662 return -1;
663
664 }
665
666 // Remove the process
667 for (uint32_t i = 0; i < m_processes.size(); i++) {
668 if (m_processes[i] == process) {
669 m_processes.erase(m_processes.begin() + i);
670
671 // Delete the process mem
672 delete process;
673 return 0;
674 }
675 }
676
677 // Process not found
678 return -1;
679}

687cpu_status_t* Scheduler::force_remove_process(Process* process) {
688
689 // If there is no process, fail
690 if (!process)
691 return nullptr;
692
693 // Remove all the threads
694 for (auto thread: process->threads()) {
695
696 // Remove the thread from the scheduler
697 size_t index = m_threads.find(thread) - m_threads.begin();
698 m_threads.erase(m_threads.begin() + index);
699
700 // Delete the thread
701 process->remove_thread(thread->tid);
702
703 }
704
705 // Process will be dead now so run the next process (don't care about the execution state being outdated as it is being
706 // removed regardless)
707 return schedule_next(&current_thread()->execution_state);
708}
