117 void iterate_entries(const void* buffer, size_t size, iterate_entry_callback_t cb) {
118
119 // Iterate each entry
120 size_t offset = 0;
121 while (offset + sizeof(entry_information_t) <= size) {
122
123 // Get the entry
124 auto* entry = (const entry_information_t*)(buffer + offset);
125
126 // Check bounds
127 if (entry->entry_length < sizeof(entry_information_t) || offset + entry->entry_length > size)
128 break;
129
130 // Invoke callback
131 const char* name = entry->name;
132 if (!cb(entry, name))
133 break;
134
135 // Move the next entry
136 offset += entry->entry_length;
137 }
138 }
