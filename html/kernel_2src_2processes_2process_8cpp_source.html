32Thread::Thread(void (* _entry_point)(void*), void* args, int arg_amount, Process* parent)
33{
34
35 // Basic setup
36 thread_state = ThreadState::NEW;
37 wakeup_time = 0;
38 ticks = 0;
39
40 // Create the stack ()
41 m_stack_pointer = (uintptr_t) parent->memory_manager->allocate(STACK_SIZE) + STACK_SIZE;
42
43 // Use the kernel stack
44 m_tss_stack_pointer = parent->is_kernel ? CPU::executing_core() -> tss.rsp0 : (uintptr_t) parent->memory_manager->kmalloc(STACK_SIZE) + STACK_SIZE;
45
46 // Mak sure there is a stack
47 ASSERT(m_stack_pointer != 0 && m_tss_stack_pointer != 0, "Failed to allocate stack for thread");
48
49 // Set up the execution state
50 execution_state = {};
51 execution_state.rip = (uint64_t) _entry_point;
52 execution_state.ss = parent->is_kernel ? 0x10 : 0x23;
53 execution_state.cs = parent->is_kernel ? 0x8 : 0x1B;
54 execution_state.rflags = 0x202;
55 execution_state.interrupt_number = 0;
56 execution_state.error_code = 0;
57 execution_state.rsp = m_stack_pointer;
58 execution_state.rbp = 0;
59
60 // Copy the args into userspace
61 uint64_t argc = arg_amount;
62 size_t arg_size = argc * sizeof(void*);
63 void* argv = parent->memory_manager->allocate(arg_size);
64 for (int i = 0; i < argc; i++) {
65
66 // Copy each argument
67 size_t len = strlen(((char**)args)[i]) + 1;
68 ((char**)argv)[i] = (char*) parent->memory_manager->allocate(len);
69 memcpy(((char**)argv)[i], ((char**)args)[i], len);
70 }
71
72 execution_state.rdi = argc;
73 execution_state.rsi = (uint64_t) argv;
74 //execution_state->rdx = (uint64_t)env_args;
75
76 // All ports denied by default
77 memset(m_io_bitmap, 0xFF, sizeof(m_io_bitmap));
78
79 parent_pid = parent->pid();
80}

103void Thread::yield() {
104
105 // Only update the state if the caller hasn't specified the resume condition state
106 if (thread_state == ThreadState::RUNNING)
107 thread_state = ThreadState::READY;
108
109 // Save a checkpoint (resumed state is ret of the below function: execution after yield will be there)
110 save_cpu_state();
111
112 // Schedule the replacement thread, but guard against infinite loops where thread continusly pass execution to the
113 // next thread after scheduler resumes it post yield)
114 if(thread_state != ThreadState::RUNNING){
115
116 // Yield to the next thread
117 cpu_status_t* next = GlobalScheduler::core_scheduler()->schedule_next(&execution_state);
118 InterruptManager::ForceInterruptReturn(next);
119 }
