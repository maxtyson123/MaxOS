48bool IntegratedDriveElectronicsController::identify_device(uint16_t command_base, bool master) {
49
50 // Get the ports
51 auto data_port = Port16Bit(command_base);
52 auto device_port = Port8Bit(command_base + 6);
53 auto command_port = Port8Bit(command_base + 7);
54
55 // Check if the master is present
56 device_port.write(0xA0);
57 uint8_t status = command_port.read();
58 if (status == 0xFF)
59 return false;
60
61 // Select the device (master or slave)
62 device_port.write(master ? 0xA0 : 0xB0);
63
64 // Check if the device is present
65 command_port.write(0x0EC);
66 status = command_port.read();
67 if (status == 0x00)
68 return false;
69
70 // Wait for the device to be ready or for an error to occur
71 while (((status & 0x80) == 0x80) && ((status & 0x01) != 0x01))
72 status = command_port.read();
73
74 //Check for any errors
75 if (status & 0x01)
76 return false;
77
78 // Skip until the start of the name
79 uint16_t buffer[256];
80 for (int i = 0; i < 256; i++)
81 buffer[i] = data_port.read();
82
83 // Get Name of the disk
84 string name = " "s * 40;
85 for (int i = 0; i < 20; i++) {
86 name[i * 2] = (buffer[27 + i] >> 8) & 0xFF;
87 name[i * 2 + 1] = buffer[27 + i] & 0xFF;
88 }
89 name = name.strip();
90
91 // Device is present and ready
92 klog("Found %s ata on port 0x%x (%s)\n", master ? "master" : "slave", command_base, name.c_str());
93 return true;
94}
