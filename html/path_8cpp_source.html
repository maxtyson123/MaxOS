130bool Path::is_child_of(const string &path, const string &parent) {
131
132 // Validation (@todo add to all)
133 if (!valid(path) || !valid(parent))
134 return false;
135
136 // Normalise both
137 string abs_path = absolute_path(path);
138 string abs_parent = absolute_path(parent);
139
140 // Ensure the parent is a directory
141 if (is_file(abs_parent))
142 abs_parent = parent_directory(abs_parent);
143
144 // Not a parent of itself
145 if (abs_path == abs_parent)
146 return false;
147
148 // Root is parent of all
149 if (abs_parent == "/")
150 return true;
151
152 // Child would have to be longer than parent
153 if (abs_path.length() <= abs_parent.length())
154 return false;
155
156 // Child prefix must be parent
157 if (abs_path.substring(0, abs_parent.length()) != abs_parent)
158 return false;
159
160 // Prefix must end with "/" otherwise could be a longer name of the parent ie "/a/b" vs "ab/b"
161 if (abs_path[abs_parent.length()] != '/')
162 return false;
163
164 // Must be parent
165 return true;
166}

219string Path::absolute_path(const string& path) {
220
221 // Split the path into components
222 auto components = path.split("/");
223 common::Vector<string> absolute_components;
224 for (auto &component: components) {
225
226 // Skip empty or self references
227 if (component == "" || component == ".")
228 continue;
229
230 // Handle parent directory references
231 if (component == "..") {
232 if (!absolute_components.empty())
233 absolute_components.pop_back();
234 } else
235 absolute_components.push_back(component);
236 }
237
238 // Join the components back into a path
239 string absolute_path = "/";
240 for(const auto& component : absolute_components)
241 absolute_path += component + "/";
242
243 // Remove trailing slash if file
244 if(is_file(path) && absolute_path.length() > 1)
245 absolute_path = absolute_path.substring(0, absolute_path.length() - 1);
246
247 return absolute_path;
248}
