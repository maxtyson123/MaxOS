22 void* make_syscall(SyscallType type, uint64_t arg0, uint64_t arg1, uint64_t arg2, uint64_t arg3, uint64_t arg4, uint64_t arg5){
23
24 void* result;
25 asm volatile(
26 "mov %[a0], %%rdi\n\t" // arg0 -> rdi
27 "mov %[a1], %%rsi\n\t" // arg1 -> rsi
28 "mov %[a2], %%rdx\n\t" // arg2 -> rdx
29 "mov %[a3], %%r10\n\t" // arg3 -> r10
30 "mov %[a4], %%r8\n\t" // arg4 -> r8
31 "mov %[a5], %%r9\n\t" // arg5 -> r9
32 "mov %[num], %%rax\n\t" // syscall number -> rax
33 "int $0x80\n\t"
34 : "=a"(result)
35 : [num] "r"((uint64_t)type),
36 [a0] "r"(arg0),
37 [a1] "r"(arg1),
38 [a2] "r"(arg2),
39 [a3] "r"(arg3),
40 [a4] "r"(arg4),
41 [a5] "r"(arg5)
42 : "rdi", "rsi", "rdx", "r10", "r8", "r9", "memory"
43 );
44
45 return result;

54 void klog(char const *format, ...){
55
56 // Create a message buffer so that not having a syscall for each character
57 char message[1024];
58 va_list parameters;
59 va_start(parameters, format);
60
61 // Format the string
62 size_t index = 0;
63 for (; *format != '\0' && index < sizeof(message) - 1; format++) {
64
65 // If it is not a %, copy the character
66 if (*format != '%') {
67 message[index++] = *format;
68 continue;
69 }
70
71 // Move to the next character
72 format++;
73 switch (*format) {
74 case 'd': {
75 // Print a decimal
76 int number = va_arg (parameters, int);
77 char* num_str = itoa(10, number);
78
79 // Append the number string
80 for (size_t i = 0; num_str[i] != '\0' && index < sizeof(message) - 1; i++)
81 message[index++] = num_str[i];
82 break;
83 }
84 case 'x': {
85 // Print a hex
86 uint64_t number = va_arg (parameters, uint64_t);
87 char* num_str = htoa(number);
88
89 // Append the number string
90 for (size_t i = 0; num_str[i] != '\0' && index < sizeof(message) - 1; i++)
91 message[index++] = num_str[i];
92 break;
93 }
94 case 's': {
95 // Print a string
96 char* str = va_arg (parameters, char*);
97
98 for (size_t i = 0; str[i] != '\0' && index < sizeof(message) - 1; i++)
99 message[index++] = str[i];
100 break;
101 }
102
103 case 'h': {
104 // "%h" must reach the kernel untouched so it can print the message without the log header
105 message[index++] = '%';
106 if (index < sizeof(message) - 1)
107 message[index++] = 'h';
108 break;
109 }
110 }
111
112 }
113
114 // Close
115 message[index] = '\0';
116 va_end(parameters);
117
118 // Print the formatted string
119 make_syscall(SyscallType::KLOG, (uint64_t)message, 0, 0, 0, 0, 0);
120 }
