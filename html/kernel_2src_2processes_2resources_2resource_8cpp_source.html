100void BaseResourceRegistry::close_resource(Resource* resource, size_t flags) {
101
102 m_lock.lock();
103
104 // Resource isn't stored in this registry
105 if(m_resources.find(resource->name()) == m_resources.end()) {
106 m_lock.unlock();
107 return;
108 }
109
110 m_resource_uses[resource->name()]--;
111
112 // Don't close a resource that has more processes using it
113 if(m_resource_uses[resource->name()]) {
114 m_lock.unlock();
115 return;
116 }
117
118 // Can safely close the resource
119 resource->close(flags);
120 m_resources.erase(resource->name());
121 m_resource_uses.erase(resource->name());
122
123 m_lock.unlock();
124 delete resource;
125}

236uint64_t ResourceManager::open_resource(resource_type_t type, string const& name, size_t flags) {
237
238 // Make sure there is a registry of the requested type
239 auto registry = GlobalResourceRegistry::get_registry(type);
240 if(!registry)
241 return 0;
242
243 // Make sure the registry holds the requested resource

245 if(!resource)
246 return 0;
