49void BlockingLock::acquire() {
50
51 // Try to get the lock
52 for (int i = 0; (i < BLOCKING_FAST_TRY_LIMIT || must_spin()); ++i)
53 if(!__atomic_test_and_set(&m_locked, __ATOMIC_ACQUIRE))
54 return;
55
56 // Add to the queue
57 auto thread = GlobalScheduler::current_thread();
58 thread->thread_state = ThreadState::WAITING;
59 m_queue.push_back(thread->tid);
60
61 // Scheduler will awake when the process that holds this lock marks this thread as READY
62 thread->yield();
63}

68void BlockingLock::release() {
69
70 // Next thread can be run
71 if(!m_queue.empty())
72 GlobalScheduler::get_thread(m_queue.pop_front())->thread_state = ThreadState::READY;
73
74 __atomic_clear(&m_locked, __ATOMIC_RELEASE);
75}
