43uint64_t processes::exec_file(const char* name, const char* path, const char** args, size_t arg_amount, bool wait_for_completion) {
44
45 // Add the path as the first argument
46 const char** full_args = new const char*[arg_amount + 1];
47 full_args[0] = path;
48 for (size_t i = 0; i < arg_amount; i++)
49 full_args[i + 1] = args[i];
50
51 // Open the file
52 auto handle = LibFS::open_file(path);
53 if (!handle)
54 return 0;
55
56 // Load the ELF blob
57 size_t size = LibFS::file_size(handle);
58 auto buffer = new uint8_t[size];
59 LibFS::file_read(handle, buffer, size);
60
61 // Create the process
62 execute_command_t command = {
63 .args = full_args,
64 .args_count = arg_amount + 1,
65 .file_data = buffer,
66 .file_data_size = size,
67 };
68 uint64_t result = resource_create(ResourceType::PROCESS, name, 0, &command);
69
70 // Clean up
71 delete[] full_args;
72 delete[] buffer;
73 LibFS::close_file(handle);
74
75 // Wait for completion if requested @todo move this the scheduler resource create place so that it can be blocked properly
76 if (wait_for_completion && result > 0){
77
78 process_stats_t stats = get_process_stats_handle(result);
79
80 // If the process is killed it wont have a PID
81 while(stats.pid != 0) {
82 yield();
83 stats = get_process_stats_handle(result);
84 }
85
86 close_process_handle(result);
87 return stats.exit_code;
88 }
89
90 return result;
91}
