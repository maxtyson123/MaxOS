49void Manager::load_driver_list(string path) {
50
51 // Read the list of inital devices
52 auto jhandle = open_file(path.c_str());
53 auto size = file_size(jhandle);
54 auto jsonstr = string((uint8_t*)allocate_memory(size), size);
55 file_read(jhandle, jsonstr.c_str(), size);
56
57 // Parse the json
58 JSONParser parser(&jsonstr);
59 JSONNode* m_initial_drivers = parser.root();
60 parse_driver_list(m_initial_drivers);
61
62 // Clean up
63 close_file(jhandle);
64}

66void Manager::parse_driver_list(JSONNode* list) {
67
68 // Cache each driver
69 auto drivers = (*list)["drivers"s];
70 for (int i = 0; i < drivers.array_size(); ++i) {
71
72 auto& driver = drivers[i];
73 auto& device_node = driver["device"s];
74 auto& model = device_node["model"s];
75
76 // Parse
77 string name = driver["name"s];
78 string path = driver["file"s];
79 string vendor = model["vendor"s];
80 string device = model["device"s];
81 string revision = model["revision"s];
82
83 // Build the device id
84 device_identification_t id {};
85 id.model = {
86 .vendor = (uint16_t)vendor.hex_to_uint64(),
87 .device = (uint16_t)device.hex_to_uint64(),
88 .revision = (uint16_t)revision.hex_to_uint64(),
89 };
90
91 // Build the driver id
92 driver_entry_t entry {
93 .name = name,
94 .path = path,
95 };
96
97 // Cache
98 m_driver_list.insert(id, entry);
99 }
100}

102void Manager::start_driver(Device* device) {
103
104 ASSERT(device != nullptr, "Cant start a driver without an associated device");
105
106 // Driver already started
107 if (device->driver_started)
108 return;
109
110 // Get the driver details for this device
111 auto it = m_driver_list.find(device->id_info());
112 if (it == m_driver_list.end())
113 return;
114
115 // Pass the ID
116 const char** args = new const char*[1];
117 string id_str(device->id);
118 args[0] = id_str.c_str();
119
120 // Start the driver
121 auto driver = it->second;
122 exec_file(driver.name.c_str(), driver.path.c_str(), args, 1);
123 delete[] args;
124
125 device->driver_started = true;
126 klog("Started driver: %s (%s) for device %d\n", driver.name.c_str(), driver.path.c_str(), device->id);
127}
