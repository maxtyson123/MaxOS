37void UBSanHandler::print_type_mismatch(type_mismatch_info_t* info, uintptr_t ptr) {
38
39 // Print the error
40 string msg;
41 if (ptr == 0)
42 msg = "null pointer access";
43 else if (info->alignment != 0 && UBSAN_ALIGNED(ptr, info->alignment))
44 msg = StringBuilder() << "misaligned address 0x" << ptr;
45 else
46 msg = StringBuilder() << TYPE_CHECK_KINDS[info->type_check_kind] << " address 0x" << ptr << " with insufficient space for an object of type " << info->type->name;
47
48 // Print the location
49 handle(info->location, msg.c_str());
50}
