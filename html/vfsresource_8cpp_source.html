162void DirectoryResource::write_entries(void const* buffer, size_t size) const {
163
164 size_t amount_written = 0;
165
166 entry_information_t* entry = nullptr;
167 size_t entry_capacity = 0;
168
169 auto write_single_entry = [&](const string& name, size_t entry_size, bool is_file) {
170
171 // Make sure there is enough space
172 size_t required_size = sizeof(entry_information_t) + name.length() + 1;
173 if (required_size > entry_capacity) {
174 delete[] (uint8_t*)entry;
175 entry = (entry_information_t*)(new uint8_t[required_size]);
176 entry_capacity = required_size;
177 }
178
179 // Create the entry
180 entry->is_file = is_file;
181 entry->size = entry_size;
182 entry->entry_length = required_size;
183 memcpy(entry->name, name.c_str(), name.length());
184 entry->name[name.length()] = '\0';
185
186 // Not enough space
187 if (amount_written + entry->entry_length > size)
188 return false;
189
190 // Copy the entry
191 memcpy((uint8_t*)buffer + amount_written, entry, entry->entry_length);
192 amount_written += entry->entry_length;
193 return true;
194 };
195
196 // Write files
197 for (const auto& file : directory->files()) {
198 if (!write_single_entry(file->name(), file->size(), true))
199 break;
200 }
201
202 // Write directories
203 for (const auto& dir : directory->subdirectories()) {
204 if (!write_single_entry(dir->name(), dir->size(), false))
205 break;
206 }
207
208 delete[] (uint8_t*)entry;
209}

436string VFSResourceServer::process_relative_path(uint64_t pid, const string& relative_path) {
437
438 // Simplify the path
439 string path = Path::absolute_path(relative_path);
440 string working_path = "/";
441
442 // Try get the working directory
443 auto working_directory = m_working_directories.find(pid);
